local function dragToPosition(model, targetCF)

            rep.Interaction.ClientIsDragging:FireServer(model, DRAG_SECRET)
            model:SetPrimaryPartCFrame(targetCF)
        end)
        task.wait(dragInterval)
    end
    return true

local function loadWeaponConfig(name)
    local module = rep:FindFirstChild("LoadedAssets") and rep.LoadedAssets:FindFirstChild(name.."AxeClass")
    if not module then return nil end
    local ok, axeClass = pcall(require, module)
    if not ok then return nil end
    local inst = type(axeClass.new)=="function" and (axeClass.new() or axeClass.new({})) or axeClass
    return { Damage = inst.Damage or 10000, SwingCooldown = inst.SwingCooldown or 0.3, SpecialTrees = inst.SpecialTrees or {} }
end
local function selectBestWeapon(treeClass)
    local toolFolder = lp:FindFirstChild("ToolFolder")
    if not toolFolder then return nil, nil, "未找到 ToolFolder" end
    local bestWeapon, bestDamage, bestConfig = nil, -1, nil
    for _, child in pairs(toolFolder:GetChildren()) do
        local config = loadWeaponConfig(child.Name)
        if config then
            local damage = (config.SpecialTrees[treeClass] and config.SpecialTrees[treeClass].Damage) or config.Damage
            if damage > bestDamage then bestWeapon, bestDamage, bestConfig = child.Name, damage, config end
        end
    end
    return bestWeapon, bestConfig, bestDamage
end
local function getBestAxeData(treeClass)
    local toolFolder = lp:FindFirstChild("ToolFolder")
    if not toolFolder then return nil, nil, "未找到 ToolFolder" end
    local bestName, bestDamage, bestConfig = nil, -1, nil
    for _, child in pairs(toolFolder:GetChildren()) do
        local config = loadWeaponConfig(child.Name)
        if config then
            local damage = config.Damage or 0
            if treeClass and config.SpecialTrees and config.SpecialTrees[treeClass] then
                damage = config.SpecialTrees[treeClass].Damage or damage
            end
            if damage > bestDamage then bestDamage = damage; bestName = child.Name; bestConfig = config end
        end
    end
    if not bestName then return nil, nil, "未找到可用的斧头配置" end
    local toolObj = toolFolder:FindFirstChild(bestName)
    local tool = toolObj and (toolObj:IsA("Tool") and toolObj or toolObj.Value)
    if not tool then return nil, nil, "无法获取工具实例" end
    return tool, bestConfig, nil
end
local function findUncutTree(treeClass)
    local playerPos = lp.Character and lp.Character.HumanoidRootPart.Position
    if not playerPos then return nil end
    local nearestDist, nearestWood, nearestTree = math.huge, nil, nil
    for _, region in pairs(ws:GetChildren()) do
        if region.Name == "TreeRegion" then
            for _, tree in pairs(region:GetChildren()) do
                if tree:IsA("Model") and tree:FindFirstChild("TreeClass") and tree.TreeClass.Value == treeClass then
                    local woodSections = {}
                    for _, child in pairs(tree:GetChildren()) do
                        if child.Name == "WoodSection" and child:IsA("BasePart") then table.insert(woodSections, child) end
                    end
                    if #woodSections >= 2 then
                        local dist = (woodSections[1].Position - playerPos).Magnitude
                        if dist < nearestDist then nearestDist, nearestWood, nearestTree = dist, woodSections[1], tree end
                    end
                end
            end
        end
    end
    return nearestWood, nearestTree
end
local function getTreeClassFromModel(treeModel)
    local tc = treeModel:FindFirstChild("TreeClass")
    if tc then return tc.Value end
    local name = treeModel.Name
    for _, suffix in ipairs({"SpookyNeon","Spooky","GoldSwampy","Cherry","CaveCrawler","MuckySewer"}) do
        if name:find(suffix) then return suffix end
    end
    return "Generic"
end

local function bringTreeRemote(treeClass)
    local weaponName, weaponConfig, damage = selectBestWeapon(treeClass)
    if not weaponName then notify("誓言", "自动选武器失败", 3); return false end
    local req = {LoveCave=10000000, Shine=80000000, Magma=5000000, Ice=8000000, Radioactive=10000000,
                 Scale=40000000, Void=5000000, Flame=10000000, BlueFlame=3500000, RainbowFlame=3500000, Celestial=10000000}
    if req[treeClass] and damage < req[treeClass] then
        notify("誓言", string.format("伤害不足！需要至少 %.0f", req[treeClass]), 4); return false
    end
    local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown
    local woodSec, tree = findUncutTree(treeClass)
    if not woodSec then notify("誓言", "附近没有可砍的树", 3); return false end
    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local tf = lp:FindFirstChild("ToolFolder")
        if tf then
            local wo = tf:FindFirstChild(weaponName)
            tool = wo and (wo:IsA("Tool") and wo or wo.Value) or nil
        end
    end
    if not tool then notify("誓言", "未找到工具: "..weaponName, 3); return false end
    local cutEvent = tree:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无法获取CutEvent", 3); return false end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local oldCamSubj = workspace.CurrentCamera.CameraSubject
    workspace.CurrentCamera.CameraSubject = tree
    local fallenLog, existingLogs = nil, {}
    for _, log in pairs(ws.LogModels:GetChildren()) do existingLogs[log] = true end
    local treeCenter = woodSec.Position
    local stopCutting, timeoutReached = false, false
    local sendTask = task.spawn(function()
        while not stopCutting do
            pcall(remote.FireServer, remote, cutEvent, {
                height=0.3, faceVector=Vector3.new(-1,0,0), cooldown=cooldown,
                sectionId=1, hitPoints=damage, tool=tool, cuttingClass="Axe"
            })
            task.wait(0.01)
        end
    end)
    local timeoutTask = task.spawn(function()
        task.wait(5)
        if not stopCutting then timeoutReached=true; stopCutting=true end
    end)
    while not fallenLog and not stopCutting do
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not existingLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value==lp
            and log:FindFirstChild("TreeClass") and log.TreeClass.Value==treeClass then
                if (log:GetPivot().Position - treeCenter).Magnitude < 100 then fallenLog = log; break end
            end
        end
        task.wait(0.01)
    end
    stopCutting = true
    task.cancel(sendTask); task.cancel(timeoutTask)
    workspace.CurrentCamera.CameraSubject = oldCamSubj
    if timeoutReached or not fallenLog then notify("誓言", "砍树超时", 3); return false end
    local primary = fallenLog.PrimaryPart or fallenLog:FindFirstChild("WoodSection") or fallenLog:FindFirstChild("Main") or fallenLog:FindFirstChildWhichIsA("BasePart")
    if not primary then notify("誓言", "无法定位木头", 3); return false end
    if not fallenLog.PrimaryPart then fallenLog.PrimaryPart = primary end
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then notify("誓言", "无法获取玩家位置", 3); return false end
    pcall(rep.Interaction.ClientInteracted.FireServer, rep.Interaction, primary, 'Click'); task.wait(0.01)
    safeTeleport(primary.CFrame + Vector3.new(0,0,0)); task.wait(0.1)
    for _=1,3 do pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.01) end
    local oldCam = workspace.CurrentCamera.CameraSubject
    local restored = false
    local function restoreCamera() if not restored then restored=true; workspace.CurrentCamera.CameraSubject = oldCam end end
    local timeoutId = task.spawn(function() task.wait(3); restoreCamera(); notify("誓言", "拖拽超时", 3) end)
    local stopFollow = false
    local followConn = RS.Heartbeat:Connect(function()
        if not stopFollow and fallenLog and fallenLog.PrimaryPart and lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") then
            lp.Character.HumanoidRootPart.CFrame = fallenLog.PrimaryPart.CFrame
        end
    end)
    for i = 1, dragAttempts do
        pcall(function()
            rep.Interaction.ClientIsDragging:FireServer(fallenLog, DRAG_SECRET)
            fallenLog:SetPrimaryPartCFrame(playerOriginalCF)
        end)
        task.wait(dragInterval)
    end
    stopFollow = true
    if followConn then followConn:Disconnect() end
    pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.01)
    pcall(rep.Interaction.ClientInteracted.FireServer, rep.Interaction, primary, 'Click')
    safeTeleport(playerOriginalCF)
    restoreCamera()
    task.cancel(timeoutId)
    return true
end

local selectedTree, selectedSawmill, sawPos = nil, nil, nil
local isDecomposeActive = false
local decomposeConn = nil
local function dragLogToPosition(log, targetCF)

    if not primary then return false end

local function decomposeAndProcess(treeModel, sawmill, sawmillPos)
    if not sawmillPos then notify("誓言", "锯木机位置无效", 3); return end
    local sections, rootSection = {}, nil
    for _, child in pairs(treeModel:GetChildren()) do
        if child.Name == "WoodSection" and child:IsA("BasePart") and child:FindFirstChild("ID") then
            local id = child.ID.Value
            if id >= 2 then table.insert(sections, {id = id, part = child})
            elseif id == 1 then rootSection = child end
        end
    end
    if #sections == 0 then notify("誓言", "该树没有可分解的部分", 3); return end
    table.sort(sections, function(a, b) return a.id > b.id end)
    local cutEvent = treeModel:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "该模型没有 CutEvent", 3); return end
    local treeClass = getTreeClassFromModel(treeModel)
    local tool, config, err = getBestAxeData(treeClass)
    if not tool then notify("誓言", err or "无法获取斧头数据", 3); return end
    local damage = config and config.Damage or 10000
    local cooldown = config and config.SwingCooldown or 0.3
    if config and config.SpecialTrees and config.SpecialTrees[treeClass] then
        damage = config.SpecialTrees[treeClass].Damage or damage
        cooldown = config.SpecialTrees[treeClass].SwingCooldown or cooldown
    end
    local req = {LoveCave=10000000, Shine=80000000, Magma=5000000, Ice=8000000,
                 Radioactive=10000000, Scale=40000000, Void=5000000, Flame=10000000,
                 BlueFlame=3500000, RainbowFlame=3500000, Celestial=10000000}
    if req[treeClass] and damage < req[treeClass] then
        notify("誓言", string.format("伤害不足！需要至少 %.0f，当前 %.0f", req[treeClass], damage), 4)
        return
    end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then notify("誓言", "无法获取玩家位置", 3); return end
    local initialLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do initialLogs[log] = true end
    local tasks = {}
    for _, sec in ipairs(sections) do
        local part = sec.part
        local sectionId = sec.id
        local stopFlag = false
        local taskId = task.spawn(function()
            while not stopFlag do
                pcall(remote.FireServer, remote, cutEvent, {
                    height=0.11, faceVector=Vector3.new(-1,0,0), cooldown=cooldown,
                    sectionId=sectionId, hitPoints=damage, tool=tool, cuttingClass="Axe"
                })
                task.wait(0.01)
                if not part or not part.Parent then break end
            end
        end)
        tasks[taskId] = true
    end
    local function hasHighIDSections()
        for _, child in pairs(treeModel:GetChildren()) do
            if child.Name == "WoodSection" and child:FindFirstChild("ID") then
                if child.ID.Value >= 2 then return true end
            end
        end
        return false
    end
    local startTime = tick()
    while hasHighIDSections() and tick() - startTime < 120 do task.wait(0.01) end
    for taskId in pairs(tasks) do task.cancel(taskId) end
    if hasHighIDSections() then notify("誓言", "停止", 3); return end
    local function getAllMyLogs()
        local logs = {}
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not initialLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp then
                table.insert(logs, log)
            end
        end
        return logs
    end
    local allLogs = getAllMyLogs()
    if #allLogs == 0 then notify("誓言", "未找到木头", 3); return end
    local targetPos = sawmillPos + Vector3.new(0.7, 0, 0)
    for idx, log in ipairs(allLogs) do
        if log and log.Parent then
            dragLogToPosition(log, targetPos)
            if idx == 1 then
                local temp = Instance.new("Part")
                temp.Size = Vector3.one; temp.Anchored = true; temp.Transparency = 1
                temp.CanCollide = false; temp.Parent = workspace
                local sP = sawmillPos + Vector3.new(0, 1, 0)
                local eP = sawmillPos + Vector3.new(3, 1, 0)
                temp.CFrame = sP
                for i = 1, 10 do
                    pcall(function()
                        rep.Interaction.ClientIsDragging:FireServer(temp, DRAG_SECRET)
                        temp.CFrame = sP:Lerp(eP, i / 10)
                    end)
                end
                for i = 1, 5 do
                    pcall(function()
                        rep.Interaction.ClientIsDragging:FireServer(temp, DRAG_SECRET)
                        temp.CFrame = eP:Lerp(sP, i / 5)
                    end)
                end
                temp:Destroy()
            end
        end
    end
    notify("誓言", "所有木头已加工", 2)
    if rootSection and rootSection.Parent then
        if not treeModel.PrimaryPart then treeModel.PrimaryPart = rootSection end
        safeTeleport(rootSection.CFrame + Vector3.new(0, 2, 0))
        dragToPosition(treeModel, targetPos)
    end
    notify("誓言", "分解完成", 3)
end
local function startDecomposeSelection()
    if isDecomposeActive then notify("誓言", "分解选择已激活", 2); return end
    isDecomposeActive = true
    notify("誓言", "请点击一棵树", 3)
    local step = 0
    decomposeConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then return end
        local isTree = false
        local hasWood = false
        for _, child in pairs(model:GetChildren()) do
            if child.Name == "WoodSection" then hasWood = true end
        end
        if hasWood and model:FindFirstChild("CutEvent") then isTree = true end
        local isSawmill = false
        local sawmillObj = nil
        local sawmillPos = nil
        local itemName = model:FindFirstChild("ItemName")
        if itemName and type(itemName.Value) == "string" and itemName.Value:find("Sawmill") then
            isSawmill = true; sawmillObj = model
        end
        if not isSawmill then
            if model.Name:find("Sawmill") or (model.Parent and model.Parent.Name:find("Sawmill")) then
                sawmillObj = model.Parent or model; isSawmill = true
            end
        end
        if isSawmill and sawmillObj then
            local particles = sawmillObj:FindFirstChild("Particles")
            if particles then sawmillPos = particles.CFrame
            else notify("誓言", "该锯木机没有 Particles", 3); return end
        end
        if step == 0 and isTree then
            selectedTree = model; step = 1
            notify("誓言", "已选中树，请点击锯木机", 3); return
        end
        if step == 1 and isSawmill and selectedTree then
            if not sawmillPos then notify("誓言", "锯木机位置无效", 3); return end
            selectedSawmill = sawmillObj; sawPos = sawmillPos
            if decomposeConn then decomposeConn:Disconnect(); decomposeConn = nil end
            isDecomposeActive = false
            notify("誓言", "开始分解...", 3)
            task.spawn(function()
                decomposeAndProcess(selectedTree, selectedSawmill, sawPos)
                selectedTree = nil; selectedSawmill = nil; sawPos = nil
            end)
            return
        end
        if step == 1 and isTree then
            selectedTree = model
            notify("誓言", "已更换选中的树", 3); return
        end
        notify("誓言", "请按顺序点击（先树后锯木机）", 2)
    end)
end

-- 拖动 & RightShift
do
    local dragging, ds, sp
    Main.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=true; ds=i.Position; sp=Main.Position
            i.Changed:Connect(function() if i.UserInputState==Enum.UserInputState.End then dragging=false end end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            local d = i.Position - ds
            Main.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
        end
    end)
    UIS.InputBegan:Connect(function(input, proc)
        if proc then return end
        if input.KeyCode == Enum.KeyCode.RightShift then OathGui.Enabled = not OathGui.Enabled end
    end)
end

local function makeDropdownWithPlus(text, options, default, order, cb)
    local HEADER_H = 32
    local SEARCH_H = 26
    local ITEM_H = 24

    local container = new("Frame",{
        Size=UDim2.new(1, 0, 0, HEADER_H),
        BackgroundColor3=T.RowBg,
        BorderSizePixel=0,
        LayoutOrder=order,
        ClipsDescendants=false,
    }, fc)
    crn(container, 6); strk(container, Color3.fromRGB(48,48,52), 1, 0.3)

    local cur = default or options[1]

    new("TextLabel",{
        Position=UDim2.new(0, 12, 0, 0),
        Size=UDim2.new(0.35, 0, 0, HEADER_H),
        BackgroundTransparency=1,
        Text=text,
        Font=Enum.Font.GothamMedium, TextSize=12,
        TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left,
    }, container)

    local valueLbl = new("TextLabel",{
        Position=UDim2.new(0.4, 0, 0, 0),
        Size=UDim2.new(0.5, -12, 0, HEADER_H),
        BackgroundTransparency=1,
        Text=tostring(cur),
        Font=Enum.Font.Gotham, TextSize=11,
        TextColor3=T.Accent, TextXAlignment=Enum.TextXAlignment.Right,
        TextTruncate=Enum.TextTruncate.AtEnd,
    }, container)

    local plusBtn = new("TextButton",{
        Size=UDim2.fromOffset(22, 22),
        Position=UDim2.new(1, -28, 0.5, -11),
        BackgroundColor3=Color3.fromRGB(48, 48, 52),
        BorderSizePixel=0,
        Text="+",
        Font=Enum.Font.GothamBold, TextSize=16,
        TextColor3=T.Text, AutoButtonColor=false,
    }, container)
    crn(plusBtn, 4)

    local listFrame = new("Frame",{
        Position=UDim2.new(0, 0, 0, HEADER_H),
        Size=UDim2.new(1, 0, 0, 0),
        BackgroundColor3=T.Panel,
        BorderSizePixel=0,
        ClipsDescendants=true,
        Visible=false,
        ZIndex=10,
    }, container)
    crn(listFrame, 6)

    local searchBox = new("TextBox",{
        Position=UDim2.new(0, 4, 0, 4),
        Size=UDim2.new(1, -8, 0, SEARCH_H - 4),
        BackgroundColor3=Color3.fromRGB(24, 25, 29),
        BorderSizePixel=0,
        Text="",
        PlaceholderText="搜索...",
        PlaceholderColor3=T.Sub,
        Font=Enum.Font.Gotham, TextSize=11,
        TextColor3=T.Text,
        TextXAlignment=Enum.TextXAlignment.Left,
        ClearTextOnFocus=false,
        ZIndex=11,
    }, listFrame)
    crn(searchBox, 4)
    new("UIPadding",{ PaddingLeft=UDim.new(0,6) }, searchBox)

    local scrollFrame = new("ScrollingFrame",{
        Position=UDim2.new(0, 4, 0, SEARCH_H + 4),
        Size=UDim2.new(1, -8, 0, 0),
        BackgroundTransparency=1,
        BorderSizePixel=0,
        ScrollBarThickness=3,
        ScrollBarImageColor3=T.Stroke,
        CanvasSize=UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize=Enum.AutomaticSize.Y,
        ZIndex=11,
    }, listFrame)
    new("UIListLayout",{ Padding=UDim.new(0, 2), SortOrder=Enum.SortOrder.LayoutOrder }, scrollFrame)

    local expanded = false
    local allOptions = {}
    for _, o in ipairs(options) do table.insert(allOptions, o) end

    local function buildItems(filterText)
        for _, ch in ipairs(scrollFrame:GetChildren()) do
            if ch:IsA("TextButton") then ch:Destroy() end
        end
        filterText = (filterText or ""):lower()
        local count = 0
        for i, opt in ipairs(allOptions) do
            local txt = tostring(opt)
            if filterText == "" or txt:lower():find(filterText, 1, true) then
                count = count + 1
                local b = new("TextButton",{
                    Size=UDim2.new(1, 0, 0, ITEM_H - 2),
                    BackgroundTransparency=1,
                    Text="   "..txt,
                    Font=Enum.Font.Gotham, TextSize=11,
                    TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left,
                    LayoutOrder=i, AutoButtonColor=false, ZIndex=12,
                }, scrollFrame)
                b.MouseEnter:Connect(function() b.BackgroundTransparency = 0.1; b.BackgroundColor3 = T.Hover end)
                b.MouseLeave:Connect(function() b.BackgroundTransparency = 1 end)
                b.MouseButton1Click:Connect(function()
                    cur = opt
                    valueLbl.Text = tostring(cur)
                    if cb then pcall(cb, cur) end
                    expand(false)
                end)
            end
        end
        return count
    end

    local function expand(state)
        expanded = state
        if state then
            plusBtn.Text = "−"
            listFrame.Visible = true
            local cnt = buildItems(searchBox.Text)
            local listH = SEARCH_H + 8 + math.min(cnt * ITEM_H, 180)
            TS:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, listH) }):Play()
            TS:Create(scrollFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, -8, 0, math.min(cnt * ITEM_H, 180)) }):Play()
            TS:Create(container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, HEADER_H + listH) }):Play()
        else
            plusBtn.Text = "+"
            TS:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 0) }):Play()
            TS:Create(container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, HEADER_H) }):Play()
            task.delay(0.2, function() listFrame.Visible = false end)
        end
    end

    plusBtn.MouseButton1Click:Connect(function() expand(not expanded) end)

    searchBox:GetPropertyChangedSignal("Text"):Connect(function()
        if expanded then
            local cnt = buildItems(searchBox.Text)
            local listH = SEARCH_H + 8 + math.min(cnt * ITEM_H, 180)
            TS:Create(listFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, listH) }):Play()
            TS:Create(scrollFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, -8, 0, math.min(cnt * ITEM_H, 180)) }):Play()
            TS:Create(container, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, HEADER_H + listH) }):Play()
        end
    end)

    return container
end
