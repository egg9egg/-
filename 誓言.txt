local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in next, props do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

-- 找最近的未砍树
local function findUncutTree(treeClass)
    local playerPos = lp.Character and lp.Character.HumanoidRootPart.Position
    if not playerPos then return nil end
    local nearestDist, nearestWood, nearestTree = math.huge, nil, nil
    for _, region in pairs(ws:GetChildren()) do
        if region.Name == "TreeRegion" then
            for _, tree in pairs(region:GetChildren()) do
                if tree:IsA("Model") and tree:FindFirstChild("TreeClass") and tree.TreeClass.Value == treeClass then
                    local woodSections = {}
                    for _, child in pairs(tree:GetChildren()) do
                        if child.Name == "WoodSection" and child:IsA("BasePart") then table.insert(woodSections, child) end
                    end
                    if #woodSections >= 2 then
                        local dist = (woodSections[1].Position - playerPos).Magnitude
                        if dist < nearestDist then nearestDist, nearestWood, nearestTree = dist, woodSections[1], tree end
                    end
                end
            end
        end
    end
    return nearestWood, nearestTree
end

-- 加载武器配置
local function loadWeaponConfig(weaponName)
    local module = rep:FindFirstChild("LoadedAssets") and rep.LoadedAssets:FindFirstChild(weaponName .. "AxeClass")
    if not module then return nil end
    local success, axeClass = pcall(require, module)
    if not success then return nil end
    local axeInstance = type(axeClass.new) == "function" and (axeClass.new() or axeClass.new({})) or axeClass
    return { Damage = axeInstance.Damage or 10000, SwingCooldown = axeInstance.SwingCooldown or 0.3, SpecialTrees = axeInstance.SpecialTrees or {} }
end

local function selectBestWeapon(treeClass)
    local toolFolder = lp:FindFirstChild("ToolFolder")
    if not toolFolder then return nil, nil, "未找到 ToolFolder" end
    local bestWeapon, bestDamage, bestConfig = nil, -1, nil
    for _, child in pairs(toolFolder:GetChildren()) do
        local config = loadWeaponConfig(child.Name)
        if config then
            local damage = (config.SpecialTrees[treeClass] and config.SpecialTrees[treeClass].Damage) or config.Damage
            if damage > bestDamage then bestWeapon, bestDamage, bestConfig = child.Name, damage, config end
        end
    end
    return bestWeapon, bestConfig, bestDamage
end

-- 带来树（核心逻辑）
local function bringTreeRemote(treeClass)
    local weaponName, weaponConfig, damage = selectBestWeapon(treeClass)
    if not weaponName then notify("誓言", "自动选武器失败", 3) return false end
    local req = { LoveCave = 10000000, Shine = 80000000, Magma = 5000000, Ice = 8000000, Radioactive = 10000000,
                  Scale = 40000000, Void = 5000000, Flame = 10000000, BlueFlame = 3500000, RainbowFlame = 3500000, Celestial = 10000000 }
    if req[treeClass] and damage < req[treeClass] then
        notify("誓言", string.format("伤害不足！需要至少 %.0f", req[treeClass]), 4)
        return false
    end
    local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown

    local woodSec, tree = findUncutTree(treeClass)
    if not woodSec then notify("誓言", "附近没有可砍的树", 3) return false end

    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local toolFolder = lp:FindFirstChild("ToolFolder")
        if toolFolder then
            local weaponObj = toolFolder:FindFirstChild(weaponName)
            tool = weaponObj and (weaponObj:IsA("Tool") and weaponObj or weaponObj.Value) or nil
        end
    end
    if not tool then notify("誓言", "未找到工具: " .. weaponName, 3) return false end

    local cutEvent = tree:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无法获取CutEvent", 3) return false end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local oldCamSubj = workspace.CurrentCamera.CameraSubject
    workspace.CurrentCamera.CameraSubject = tree
    local fallenLog, existingLogs = nil, {}
    for _, log in pairs(ws.LogModels:GetChildren()) do existingLogs[log] = true end
    local treeCenter = woodSec.Position
    local stopCutting, timeoutReached = false, false
    local sendTask = task.spawn(function()
        while not stopCutting do
            pcall(remote.FireServer, remote, cutEvent, { height = 0.3, faceVector = Vector3.new(-1, 0, 0), cooldown = cooldown, sectionId = 1, hitPoints = damage, tool = tool, cuttingClass = "Axe" })
            task.wait(0.01)
        end
    end)
    local timeoutTask = task.spawn(function() task.wait(5); if not stopCutting then timeoutReached = true; stopCutting = true end end)
    while not fallenLog and not stopCutting do
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not existingLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp and log:FindFirstChild("TreeClass") and log.TreeClass.Value == treeClass then
                if (log:GetPivot().Position - treeCenter).Magnitude < 100 then fallenLog = log; break end
            end
        end
        task.wait(0.01)
    end
    stopCutting = true; task.cancel(sendTask); task.cancel(timeoutTask)
    workspace.CurrentCamera.CameraSubject = oldCamSubj
    if timeoutReached or not fallenLog then notify("誓言", "砍树超时", 3) return false end
    local primary = fallenLog.PrimaryPart or fallenLog:FindFirstChild("WoodSection") or fallenLog:FindFirstChild("Main") or fallenLog:FindFirstChildWhichIsA("BasePart")
    if not primary then notify("誓言", "无法定位木头", 3) return false end
    if not fallenLog.PrimaryPart then fallenLog.PrimaryPart = primary end
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then notify("誓言", "无法获取玩家位置", 3) return false end
    pcall(rep.Interaction.ClientInteracted.FireServer, rep.Interaction, primary, 'Click'); task.wait(0.01)
    safeTeleport(primary.CFrame + Vector3.new(0, 0, 0)); task.wait(0.1)
    for _ = 1, 3 do pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.01) end
    local oldCam = workspace.CurrentCamera.CameraSubject
    local restored = false
    local function restoreCamera()
        if not restored then restored = true; workspace.CurrentCamera.CameraSubject = oldCam end
    end
    local timeoutId = task.spawn(function() task.wait(3); restoreCamera(); notify("誓言", "拖拽超时", 3) end)
    local stopFollow = false
    local followConn = RunService.Heartbeat:Connect(function()
        if not stopFollow and fallenLog and fallenLog.PrimaryPart and lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") then
            lp.Character.HumanoidRootPart.CFrame = fallenLog.PrimaryPart.CFrame
        end
    end)
    for i = 1, dragAttempts do
        pcall(function()
            rep.Interaction.ClientIsDragging:FireServer(fallenLog, DRAG_SECRET)
            fallenLog:SetPrimaryPartCFrame(playerOriginalCF)
        end)
        task.wait(dragInterval)
    end
    stopFollow = true
    if followConn then followConn:Disconnect() end
    pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.01)
    pcall(rep.Interaction.ClientInteracted.FireServer, rep.Interaction, primary, 'Click')
    safeTeleport(playerOriginalCF)
    restoreCamera()
    task.cancel(timeoutId)
    return true
end

-- 一键加工
local function getBestAxeData(treeClass)
    local toolFolder = lp:FindFirstChild("ToolFolder")
    if not toolFolder then return nil, nil, "未找到 ToolFolder" end
    local bestWeaponName, bestDamage, bestConfig = nil, -1, nil
    for _, child in pairs(toolFolder:GetChildren()) do
        local config = loadWeaponConfig(child.Name)
        if config then
            local damage = config.Damage or 0
            if treeClass and config.SpecialTrees and config.SpecialTrees[treeClass] then
                damage = config.SpecialTrees[treeClass].Damage or damage
            end
            if damage > bestDamage then
                bestDamage = damage; bestWeaponName = child.Name; bestConfig = config
            end
        end
    end
    if not bestWeaponName then return nil, nil, "未找到可用的斧头配置" end
    local toolObj = toolFolder:FindFirstChild(bestWeaponName)
    local tool = toolObj and (toolObj:IsA("Tool") and toolObj or toolObj.Value)
    if not tool then return nil, nil, "无法获取工具实例" end
    return tool, bestConfig, nil
end

local function getTreeClassFromModel(treeModel)
    local tc = treeModel:FindFirstChild("TreeClass")
    if tc then return tc.Value end
    local name = treeModel.Name
    for _, suffix in ipairs({ "SpookyNeon", "Spooky", "GoldSwampy", "Cherry", "CaveCrawler", "MuckySewer" }) do
        if name:find(suffix) then return suffix end
    end
    return "Generic"
end

local function decomposeAndProcess(treeModel, sawmill, sawmillPos)
    if not sawmillPos then notify("誓言", "锯木机位置无效", 3) return end
    local sections, rootSection = {}, nil
    for _, child in pairs(treeModel:GetChildren()) do
        if child.Name == "WoodSection" and child:IsA("BasePart") and child:FindFirstChild("ID") then
            local id = child.ID.Value
            if id >= 2 then table.insert(sections, { id = id, part = child })
            elseif id == 1 then rootSection = child end
        end
    end
    if #sections == 0 then notify("誓言", "该树没有可分解的部分", 3) return end
    table.sort(sections, function(a, b) return a.id > b.id end)

    local cutEvent = treeModel:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "该模型没有 CutEvent", 3) return end

    local treeClass = getTreeClassFromModel(treeModel)
    local tool, config, err = getBestAxeData(treeClass)
    if not tool then notify("誓言", err or "无法获取斧头数据", 3) return end

    local damage = config and config.Damage or 10000
    local cooldown = config and config.SwingCooldown or 0.3
    if config and config.SpecialTrees and config.SpecialTrees[treeClass] then
        damage = config.SpecialTrees[treeClass].Damage or damage
        cooldown = config.SpecialTrees[treeClass].SwingCooldown or cooldown
    end

    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then notify("誓言", "无法获取玩家位置", 3) return end

    local initialLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do initialLogs[log] = true end

    local tasks = {}
    for _, sec in ipairs(sections) do
        local part = sec.part
        local sectionId = sec.id
        local stopFlag = false
        local taskId = task.spawn(function()
            while not stopFlag do
                pcall(remote.FireServer, remote, cutEvent, {
                    height = 0.11, faceVector = Vector3.new(-1, 0, 0),
                    cooldown = cooldown, sectionId = sectionId,
                    hitPoints = damage, tool = tool, cuttingClass = "Axe"
                })
                task.wait(0.01)
                if not part or not part.Parent then break end
            end
        end)
        tasks[taskId] = true
    end

    local function hasHighIDSections()
        for _, child in pairs(treeModel:GetChildren()) do
            if child.Name == "WoodSection" and child:FindFirstChild("ID") then
                if child.ID.Value >= 2 then return true end
            end
        end
        return false
    end
    local startTime = tick()
    while hasHighIDSections() and tick() - startTime < 120 do task.wait(0.01) end
    for taskId in pairs(tasks) do task.cancel(taskId) end
    if hasHighIDSections() then notify("誓言", "停止", 3) return end

    local function getAllMyLogs()
        local logs = {}
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not initialLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp then
                table.insert(logs, log)
            end
        end
        return logs
    end
    local allLogs = getAllMyLogs()
    if #allLogs == 0 then notify("誓言", "未找到木头", 3) return end

    local targetPos = sawmillPos + Vector3.new(0.7, 0, 0)
    for idx, log in ipairs(allLogs) do
        if log and log.Parent then dragLogToPosition(log, targetPos) end
    end
    notify("誓言", "所有木头已拖到锯木机", 2)

    if rootSection and rootSection.Parent then
        if not treeModel.PrimaryPart then treeModel.PrimaryPart = rootSection end
        safeTeleport(rootSection.CFrame + Vector3.new(0, 2, 0))
        dragToPosition(treeModel, targetPos)
    end
    notify("誓言", "分解完成", 3)
end

local selectedTree = nil
local selectedSawmill = nil
local sawPos = nil
local isDecomposeActive = false
local decomposeConn = nil

local function startDecomposeSelection()
    if isDecomposeActive then notify("誓言", "分解选择已激活", 2) return end
    isDecomposeActive = true
    selectedTree = nil; selectedSawmill = nil; sawPos = nil
    notify("誓言", "请点击一棵树", 3)
    local step = 0
    decomposeConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then return end
        local hasWood = false
        for _, child in pairs(model:GetChildren()) do
            if child.Name == "WoodSection" then hasWood = true end
        end
        local isTree = hasWood and model:FindFirstChild("CutEvent")
        local isSawmill = false
        local sawmillObj = nil
        local sawmillPos = nil
        local itemName = model:FindFirstChild("ItemName")
        if itemName and type(itemName.Value) == "string" and itemName.Value:find("Sawmill") then
            isSawmill = true; sawmillObj = model
        end
        if not isSawmill then
            if model.Name:find("Sawmill") or (model.Parent and model.Parent.Name:find("Sawmill")) then
                sawmillObj = model.Parent or model; isSawmill = true
            end
        end
        if isSawmill and sawmillObj then
            local particles = sawmillObj:FindFirstChild("Particles")
            if particles then sawmillPos = particles.CFrame
            else notify("誓言", "该锯木机没有 Particles", 3); return end
        end

        if step == 0 and isTree then
            selectedTree = model; step = 1
            notify("誓言", "已选中树，请点击一个锯木机", 3)
            return
        end
        if step == 1 and isSawmill and selectedTree then
            if not sawmillPos then notify("誓言", "锯木机位置无效", 3); return end
            selectedSawmill = sawmillObj; sawPos = sawmillPos
            if decomposeConn then decomposeConn:Disconnect(); decomposeConn = nil end
            isDecomposeActive = false
            notify("誓言", "开始分解...", 3)
            task.spawn(function()
                decomposeAndProcess(selectedTree, selectedSawmill, sawPos)
                selectedTree = nil; selectedSawmill = nil; sawPos = nil
            end)
            return
        end
        if step == 1 and isTree then
            selectedTree = model
            notify("誓言", "已更换选中的树", 3)
            return
        end
        notify("誓言", "请按顺序点击（先树后锯木机）", 2)
    end)
end

                if lavaScript and (lavaScript:IsA("Script") or lavaScript:IsA("LocalScript")) then
                    pcall(function() child:Destroy() end); count = count + 1
                else
                    if child:IsA("Model") or child:IsA("Folder") or child:IsA("BasePart") then searchAndDestroy(child) end
                end
            end
        end
        searchAndDestroy(plainsMaze)
    end
    if count > 0 and not silent then notify("誓言", "已删除 " .. count .. " 个岩浆", 2) end
    return count
end
local function startAutoDelete(interval)
    if autoDeleteRunning then return end
    autoDeleteRunning = true
    autoDeleteTask = task.spawn(function()
        while autoDeleteRunning do deleteLavaObjectsOnce(true); task.wait(interval) end
    end)
    notify("誓言", "已开启自动删除岩浆", 3)
end
local function stopAutoDelete()
    if not autoDeleteRunning then return end
    autoDeleteRunning = false
    if autoDeleteTask then task.cancel(autoDeleteTask); autoDeleteTask = nil end
    notify("誓言", "已关闭自动删除岩浆", 3)
end

-- 鸭子带回
local function bringDuckc(duckName, ownOnly)
    local ducks = {}
    for _, model in pairs(ws.PlayerModels:GetChildren()) do
        if model:IsA("Model") and model.Name == duckName then
            local owner = model:FindFirstChild("Owner")
            if (ownOnly and owner and owner.Value == lp) or (not ownOnly and (not owner or owner.Value == nil)) then
                table.insert(ducks, model)
            end
        end
    end
    if #ducks == 0 then notify("誓言", ownOnly and "你没有该鸭子" or "没有无归属的该鸭子", 3) return end
    if not duckCycleIndex[duckName] then duckCycleIndex[duckName] = 1 end
    local selected = ducks[duckCycleIndex[duckName]]
    duckCycleIndex[duckName] = duckCycleIndex[duckName] % #ducks + 1

    local originalPos = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not originalPos then return end
    local primary = selected.PrimaryPart or selected:FindFirstChild("Main") or selected:FindFirstChildWhichIsA("BasePart")
    if not primary then notify("誓言", "无法定位鸭子", 3) return end
    if not selected.PrimaryPart then selected.PrimaryPart = primary end
    safeTeleport(primary.CFrame + Vector3.new(0, 2, 0))
    dragToPosition(selected, originalPos)
end

-- NoClip
local function applyNoClip(state)
    if state then
        if bai.noclipConn then bai.noclipConn:Disconnect() end
        bai.noclipConn = RunService.Stepped:Connect(function()
            if lp.Character then
                for _, v in pairs(lp.Character:GetChildren()) do
                    if v:IsA("BasePart") then v.CanCollide = false end
                end
            end
        end)
        if bai.noclipCharConn then bai.noclipCharConn:Disconnect() end
        bai.noclipCharConn = lp.CharacterAdded:Connect(function()
            if bai.noclipEnabled then
                task.wait(0.1)
                for _, v in pairs(lp.Character:GetChildren()) do
                    if v:IsA("BasePart") then v.CanCollide = false end
                end
            end
        end)
    else
        if bai.noclipConn then bai.noclipConn:Disconnect(); bai.noclipConn = nil end
        if bai.noclipCharConn then bai.noclipCharConn:Disconnect(); bai.noclipCharConn = nil end
        if lp.Character then
            for _, v in pairs(lp.Character:GetChildren()) do
                if v:IsA("BasePart") then v.CanCollide = true end
            end
        end
    end
    bai.noclipEnabled = state
end

local espEnabled = false

local function updateEsp()
    if not espEnabled then return end
    local cam = workspace.CurrentCamera
    local localPos = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not localPos then return end
    for p, data in pairs(espFrames) do
        local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local pos, onScreen = cam:WorldToScreenPoint(hrp.Position)
            if onScreen then
                data.frame.Visible = true; data.frame.Position = UDim2.new(0, pos.X - 25, 0, pos.Y - 70)
                data.text.Text = string.format("%.1f", (hrp.Position - localPos.Position).Magnitude)
            else data.frame.Visible = false end
        else data.frame.Visible = false end
    end
end

    if espEnabled then return end
    espEnabled = true
    espGui = Instance.new("ScreenGui"); espGui.Name = "ESP_GUI_誓言"; espGui.ResetOnSpawn = false; espGui.Parent = PlayerGui
    espFrames = {}
    for _, p in pairs(Players:GetPlayers()) do if p ~= lp then createEspFrame(p) end end
    espRenderConn = RunService.RenderStepped:Connect(updateEsp)
    espPlayerAddedConn = Players.PlayerAdded:Connect(function(p) if p ~= lp then createEspFrame(p) end end)
    espPlayerRemovingConn = Players.PlayerRemoving:Connect(removeEspFrame)
    notify("誓言", "ESP已开启", 2)
end
local function disableEsp()

end

local duckOutlineEnabled = false
local duckHighlights = {}
local duckRefreshTask = nil
local function addDuckOutline(duckModel)
    if not duckOutlineEnabled then return end
    if duckHighlights[duckModel] then return end
    local hl = Instance.new("Highlight")
    hl.Parent = duckModel
    hl.FillColor = Color3.fromRGB(0, 150, 255)
    hl.FillTransparency = 0.7
    hl.OutlineColor = Color3.fromRGB(0, 100, 200)
    hl.OutlineTransparency = 0.3
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Enabled = true
    duckHighlights[duckModel] = hl
end
local function removeDuckOutline(duckModel)
    local hl = duckHighlights[duckModel]
    if hl then pcall(function() hl:Destroy() end); duckHighlights[duckModel] = nil end
end
local function syncDuckHighlights()
    if not duckOutlineEnabled then return end
    local currentDucks = {}
    for _, model in pairs(workspace.PlayerModels:GetChildren()) do
        if model:IsA("Model") and model.Name == "2018CGift_Duck" then
            currentDucks[model] = true
            if not duckHighlights[model] then addDuckOutline(model) end
        end
    end
    for model, _ in pairs(duckHighlights) do
        if not currentDucks[model] then removeDuckOutline(model) end
    end
end
local function enableDuckOutline()
    if duckOutlineEnabled then return end
    duckOutlineEnabled = true
    if type(duckHighlights) ~= "table" then duckHighlights = {} end
    syncDuckHighlights()
    if duckRefreshTask then task.cancel(duckRefreshTask) end
    duckRefreshTask = task.spawn(function()
        while duckOutlineEnabled do
            syncDuckHighlights()
            task.wait(2)
        end
    end)
    notify("誓言", "🔵 礼物鸭透视已开启", 2)
end
local function disableDuckOutline()
    if not duckOutlineEnabled then return end
    duckOutlineEnabled = false
    if duckRefreshTask then task.cancel(duckRefreshTask); duckRefreshTask = nil end
    for _, hl in pairs(duckHighlights) do pcall(function() hl:Destroy() end) end
    duckHighlights = {}
    notify("誓言", "🔵 礼物鸭透视已关闭", 2)
end

    local myLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do

        if owner and owner.Value == lp then table.insert(myLogs, log) end
    end
    if #myLogs == 0 then notify("誓言", "没有你的木头", 3) return end
    for i, log in ipairs(myLogs) do

        if primary then
            if not log.PrimaryPart then log.PrimaryPart = primary end
            safeTeleport(primary.CFrame + Vector3.new(0, 3, 0)); task.wait(0.1)
            for _ = 1, 3 do pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.05) end

                pcall(function()

                task.wait(dragInterval)
            end
            pcall(rep.Interaction.ClientInteracted.FireServer, rep.Interaction, primary, 'Click')
            safeTeleport(originalPos); task.wait(0.1)
        end
    end

            Size = UDim2.new(ratio, 0, 1, 0),

            BorderSizePixel = 0,
        }, track)

            Size = UDim2.fromOffset(12, 12),

            BackgroundColor3 = Color3.fromRGB(240, 240, 245),

            Size = UDim2.new(1, 16, 3, 16),
            Position = UDim2.new(0, -8, 0.5, 0),
            AnchorPoint = Vector2.new(0, 0.5),
            BackgroundTransparency = 1,
            Text = "",
            ZIndex = 3,
        }, track)
        local dragging, focused = false, false
        local currentValue = default
        local function setValue(value, updateBox)
            value = math.clamp(math.floor(value + 0.5), minValue, maxValue)
            currentValue = value
            local rel = (value - minValue) / span
            fill.Size = UDim2.new(rel, 0, 1, 0)
            knob.Position = UDim2.new(rel, 0, 0.5, 0)
            if updateBox and not focused then valueBox.Text = tostring(value) end
            if onChange then onChange(value) end
        end
        setValue(default, false)
        local function applyFromX(x)
            local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
            setValue(minValue + rel * span, true)
        end
        hit.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true; applyFromX(input.Position.X)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                applyFromX(input.Position.X)
            end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
        valueBox.Focused:Connect(function() focused = true end)
        valueBox.FocusLost:Connect(function()
            focused = false
            local num = tonumber(valueBox.Text)
            if not num then valueBox.Text = tostring(currentValue); return end
            setValue(num, true)
        end)
    end

            if s then enableEsp() else disableEsp() end

            TextXAlignment = Enum.TextXAlignment.Left,

            local players = Players:GetPlayers()
            playerIndex = (playerIndex % #players) + 1
            selectedPlayer = players[playerIndex].Name

        end)

            if target and target.Character then

            end
        end)

                    safeTeleport(v.OriginSquare.CFrame + Vector3.new(0, 10, 0))

                end

    --  分区 3：脚本
    --=====================================================
    do
        local page = Pages[3]
        local container = makeFolderSection(page, "快捷操作", 1, 76)
        makeActionButton(container, "重置角色", 1, function()
            local hum = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid")
            if hum then hum.Health = 0 end

            Font = Enum.Font.Gotham,

            TextColor3 = Theme.Sub,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextWrapped = true,
        }, tip)

    --  分区 4：设置
    --=====================================================
    do
        local page = Pages[4]
        local container = makeFolderSection(page, "界面设置", 1, 52)
        makeSliderCard(container, "界面缩放 (%)", 50, 150, 100, 1, function(v) uiScale.Scale = v / 100 end)
    end

    --=====================================================
    --  分区 5：关于
    --=====================================================
    do
        local page = Pages[5]
        local card = new("Frame", {
            Size = UDim2.new(1, 0, 0, 0),
            AutomaticSize = Enum.AutomaticSize.Y,
            BackgroundColor3 = Theme.Panel,
            BorderSizePixel = 0,
            LayoutOrder = 1,
        }, page)
        corner(card, 10)
        new("UIPadding", {
            PaddingTop = UDim.new(0, 14),
            PaddingBottom = UDim.new(0, 14),
            PaddingLeft = UDim.new(0, 14),
            PaddingRight = UDim.new(0, 14),
        }, card)
        new("UIListLayout", {
            Padding = UDim.new(0, 8),
            SortOrder = Enum.SortOrder.LayoutOrder,
        }, card)
        new("TextLabel", {
            Size = UDim2.new(1, 0, 0, 24),

            Font = Enum.Font.GothamBold,

                BackgroundTransparency = 1,

                Font = Enum.Font.GothamBold,

                TextColor3 = Theme.Text,
                TextXAlignment = Enum.TextXAlignment.Right,

        addInfo("QQ 群", "1109682737", 4)
        addInfo("当前版本", VERSION, 5)
        addInfo("专用游戏", GAME_TITLE, 6)
    end

    --=====================================================
    --  拖动 & 快捷键
    --=====================================================
    do
        local dragging, dragStart, startPos
        Main.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = Main.Position
                local conn
                conn = input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                        conn:Disconnect()
                    end
                end)
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end)
        UserInputService.InputBegan:Connect(function(input, processed)
            if processed then return end
            if input.KeyCode == Enum.KeyCode.RightShift then
                if gui.Enabled and Main.Visible then closeUI()
                else gui.Enabled = true; Island.Visible = false; Main.Visible = true end
            end
        end)
    end

    selectTab(1)
end

        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset = true,
        DisplayOrder = 200,

        AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = KeyTheme.Card,
        BorderSizePixel = 0,

        Size = UDim2.new(0, 60, 0, 3),
        Position = UDim2.new(0.5, 0, 0, 12),
        AnchorPoint = Vector2.new(0.5, 0),
        BackgroundColor3 = KeyTheme.Accent,
        BorderSizePixel = 0,
        ZIndex = 3,
    }, Card)
    corner(accent, 2)

    local Title = new("TextLabel", {
        Size = UDim2.new(1, -40, 0, 56),
        Position = UDim2.new(0, 20, 0, 24),
        BackgroundTransparency = 1,
        Text = "Oath",
        Font = Enum.Font.SourceSansItalic,
        TextSize = 46,
        TextColor3 = KeyTheme.Accent,
        TextXAlignment = Enum.TextXAlignment.Center,
        ZIndex = 3,
    }, Card)

    local Subtitle = new("TextLabel", {
        Size = UDim2.new(1, -40, 0, 20),
        Position = UDim2.new(0, 20, 0, 80),
        BackgroundTransparency = 1,
        Text = "Enter your key to continue",
        Font = Enum.Font.Gotham,
        TextSize = 12,
        TextColor3 = KeyTheme.Sub,
        TextXAlignment = Enum.TextXAlignment.Center,
        ZIndex = 3,
    }, Card)

    local InputHolder = new("Frame", {
        Size = UDim2.new(1, -60, 0, 42),
        Position = UDim2.new(0.5, 0, 0, 118),
        AnchorPoint = Vector2.new(0.5, 0),
        BackgroundColor3 = KeyTheme.Input,
        BorderSizePixel = 0,
        ZIndex = 3,
    }, Card)
    corner(InputHolder, 12)
    local InputStroke = stroke(InputHolder, KeyTheme.Stroke, 1)

    local InputBox = new("TextBox", {
        Size = UDim2.new(1, -24, 1, 0),
        Position = UDim2.new(0, 12, 0, 0),
        BackgroundTransparency = 1,
        Text = "",
        PlaceholderText = "请输入卡密...",
        PlaceholderColor3 = KeyTheme.Sub,
        Font = Enum.Font.GothamMedium,
        TextSize = 14,
        TextColor3 = KeyTheme.Text,
        TextXAlignment = Enum.TextXAlignment.Left,
        ClearTextOnFocus = false,
        ZIndex = 3,

        Position = UDim2.new(0, 20, 0, 224),
        BackgroundTransparency = 1,
        Text = "",
        Font = Enum.Font.Gotham,
        TextSize = 11,
        TextColor3 = KeyTheme.Red,
        TextXAlignment = Enum.TextXAlignment.Center,
        ZIndex = 3,
    }, Card)

    InputBox.Focused:Connect(function()
        TweenService:Create(InputStroke, TweenInfo.new(0.15), { Color = KeyTheme.Accent }):Play()
    end)
    InputBox.FocusLost:Connect(function()
        TweenService:Create(InputStroke, TweenInfo.new(0.15), { Color = KeyTheme.Stroke }):Play()

    Card.Size = UDim2.fromOffset(300, 330)
    Card.BackgroundTransparency = 1
    Title.TextTransparency = 1
    Subtitle.TextTransparency = 1
    InputHolder.BackgroundTransparency = 1
    ConfirmBtn.BackgroundTransparency = 1

    TweenService:Create(Card, TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {

        BackgroundTransparency = 0,
    }):Play()
    TweenService:Create(Title, TweenInfo.new(0.4), { TextTransparency = 0 }):Play()
    TweenService:Create(Subtitle, TweenInfo.new(0.4), { TextTransparency = 0 }):Play()
    TweenService:Create(InputHolder, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()
    TweenService:Create(ConfirmBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()

    local function shake()
        local original = Card.Position
        local offsets = { 10, -10, 8, -8, 6, -6, 4, -4, 2, -2, 0 }
        for _, o in ipairs(offsets) do
            TweenService:Create(Card, TweenInfo.new(0.04), { Position = original + UDim2.fromOffset(o, 0) }):Play()
            task.wait(0.04)
        end
        Card.Position = original
    end

    local verifying = false
    local function verify()
        if verifying then return end
        verifying = true

        if KEYS[InputBox.Text] then
            TipLabel.TextColor3 = KeyTheme.Accent
            TipLabel.Text = "验证成功，正在加载..."
            local ok, err = pcall(buildMainPanel)
            if ok then
                TweenService:Create(Card, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
                    Size = UDim2.fromOffset(280, 320), BackgroundTransparency = 1,
                }):Play()
                TweenService:Create(Title, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
                TweenService:Create(Subtitle, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
                TweenService:Create(InputHolder, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
                TweenService:Create(ConfirmBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
                TweenService:Create(Backdrop, TweenInfo.new(0.35), { BackgroundTransparency = 1 }):Play()
                task.wait(0.35)
                keyGui:Destroy()
            else
                TipLabel.TextColor3 = KeyTheme.Red
                TipLabel.Text = "主脚本加载失败"
                warn("[Oath] " .. tostring(err))
                verifying = false
            end
        else
            TipLabel.TextColor3 = KeyTheme.Red
            TipLabel.Text = "卡密错误，请重试"
            shake()
            InputBox.Text = ""
            verifying = false
        end
    end

    ConfirmBtn.MouseButton1Click:Connect(verify)
    InputBox.FocusLost:Connect(function(enterPressed) if enterPressed then verify() end end)

    task.wait(0.6)
    pcall(function() InputBox:CaptureFocus() end)
    print("[Oath] 卡密验证界面已创建")
end

showKeyUI()
