local lockPlayerConn = nil
local function uprightCFrame(hrp, cf)
    local pos = cf.Position
    local yRot = hrp and hrp.Orientation.Y or 0
    return CFrame.new(pos) * CFrame.Angles(0, math.rad(yRot), 0)
end
local function unlockPlayer()
    if lockPlayerConn then lockPlayerConn:Disconnect(); lockPlayerConn = nil end
end
local function lockPlayerAt(cf)
    unlockPlayer()
    local function hold()
        local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.CFrame = uprightCFrame(hrp, cf)
            hrp.Velocity = Vector3.new(0,0,0)
            hrp.RotVelocity = Vector3.new(0,0,0)
        end
    end
    hold()
    lockPlayerConn = RS.Heartbeat:Connect(hold)
end
local function safeTeleport(cf)
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.CFrame = uprightCFrame(hrp, cf)
        task.wait(0.05)
        if hrp.Position.Y < -100 then hrp.CFrame = uprightCFrame(hrp, cf) + Vector3.new(0,10,0) end
    end
end

-- ==================== 拖拽 ====================
local function getNextDragID() bai.dragIDCounter = bai.dragIDCounter + 1; return bai.dragIDCounter end
local DRAG_TOTAL, DRAG_SET_START, DRAG_SET_COUNT = 60, 28, 2
local DRAG_SET_INTERVAL, DRAG_REFRESH_INTERVAL, DRAG_AFTER_WAIT = 0.01, 0.01, 0.01

local function dragToPosition(model, targetCF, callback, waitFinish)
    if not model or not model.Parent then if callback then callback() end; return end
    local primary = model.PrimaryPart or model:FindFirstChild("WoodSection") or model:FindFirstChild("Main") or model:FindFirstChildWhichIsA("BasePart")
    if not primary then if callback then callback() end; return end
    if not model.PrimaryPart then model.PrimaryPart = primary end
    lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))
    local dragID = getNextDragID()
    local remote = rep.Interaction.ClientIsDragging
    pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
    pcall(function() remote:FireServer("Begin", model, dragID) end)
    local lastSet = DRAG_SET_START + DRAG_SET_COUNT - 1
    for i = 1, lastSet do
        if not model.Parent or not model.PrimaryPart then
            pcall(function() remote:FireServer("End", model, dragID) end)
            if callback then callback() end; return
        end
        local shouldSet = (i >= DRAG_SET_START)
        pcall(function()
            remote:FireServer("Refresh", model, dragID)
            if shouldSet then
                rep.Interaction.ClientRequestOwnership:FireServer(primary)
                model:SetPrimaryPartCFrame(targetCF)
            end
        end)
        if i < lastSet then
            if shouldSet then task.wait(DRAG_SET_INTERVAL) else task.wait(DRAG_REFRESH_INTERVAL) end
        end
    end
    task.wait(DRAG_AFTER_WAIT)
    local dragEnded = false
    task.spawn(function()
        for i = lastSet + 1, DRAG_TOTAL do
            if not model.Parent or not model.PrimaryPart then break end
            pcall(function() remote:FireServer("Refresh", model, dragID) end)
            if i < DRAG_TOTAL then task.wait(DRAG_REFRESH_INTERVAL) end
        end
        pcall(function() remote:FireServer("End", model, dragID) end)
        dragEnded = true
    end)
    if waitFinish then
        local w = 0
        while w < 1.5 and not dragEnded do task.wait(0.05); w = w + 0.05 end
    end
    task.wait(0.4)
    local verify = 0
    while model.Parent and model.PrimaryPart and verify < 2 do
        local dist = (model.PrimaryPart.Position - targetCF.Position).Magnitude
        if dist <= 3 then break end
        verify = verify + 1
        lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))
        task.wait(0.15)
        local retryID = getNextDragID()
        pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
        pcall(function() remote:FireServer("Begin", model, retryID) end)
        for _ = 1, 10 do
            if not model.Parent then break end
            pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
            pcall(function() remote:FireServer("Refresh", model, retryID) end)
            pcall(function() model:SetPrimaryPartCFrame(targetCF) end)
            task.wait(0.05)
        end
        pcall(function() remote:FireServer("End", model, retryID) end)
        task.wait(0.15)
    end
    unlockPlayer()
    if callback then callback() end
end

local function readTreeClasses()
    local list = {}
    local amounts = ws:FindFirstChild("Stores") and ws.Stores:FindFirstChild("PlanterStore") and ws.Stores.PlanterStore:FindFirstChild("Planter Spawner") and ws.Stores.PlanterStore["Planter Spawner"]:FindFirstChild("Amounts")
    if amounts then
        for _, c in ipairs(amounts:GetChildren()) do
            if c:IsA("NumberValue") or c:IsA("IntValue") or c:IsA("StringValue") then table.insert(list, tostring(c.Name)) end
        end
    end
    if #list == 0 then
        local W = rep:FindFirstChild("Woods")
        if W then for _, c in ipairs(W:GetChildren()) do table.insert(list, tostring(c.Name)) end end
    end
    return list
end

local TREE_PRIORITY = {"火焰木","石头木","蓝色火焰木","彩色火焰木"}
local function sortTreeNames(list)
    local prio = {}
    for i, n in ipairs(TREE_PRIORITY) do prio[n] = i end
    table.sort(list, function(a, b)
        local pa, pb = prio[a], prio[b]
        if pa and pb then return pa < pb end
        if pa then return true end
        if pb then return false end
        return a < b
    end)
end
local treeNames = {}
for name in pairs(treeMapping) do table.insert(treeNames, name) end

local function findUncutTree(treeClass)
    local playerPos = lp.Character and lp.Character.HumanoidRootPart.Position
    if not playerPos then return nil end
    local nearestDist, nearestWood, nearestTree = math.huge, nil, nil
    for _, region in pairs(ws:GetChildren()) do
        if region.Name == "TreeRegion" then
            for _, tree in pairs(region:GetChildren()) do
                if tree:IsA("Model") and tree:FindFirstChild("TreeClass") and tree.TreeClass.Value == treeClass then
                    local woodSections = {}
                    for _, child in pairs(tree:GetChildren()) do
                        if child.Name == "WoodSection" and child:IsA("BasePart") then table.insert(woodSections, child) end
                    end
                    if #woodSections >= 2 then
                        local dist = (woodSections[1].Position - playerPos).Magnitude
                        if dist < nearestDist then nearestDist, nearestWood, nearestTree = dist, woodSections[1], tree end
                    end
                end
            end
        end
    end
    return nearestWood, nearestTree
end
local function getTreeTopCFrame(treeModel)
    local topY, topCF = -math.huge, nil
    for _, child in pairs(treeModel:GetChildren()) do
        if child:IsA("BasePart") then
            local top = child.Position.Y + child.Size.Y / 2
            if top > topY then topY = top; topCF = CFrame.new(child.Position.X, top + 2, child.Position.Z) end
        end
    end
    if not topCF then topCF = treeModel:GetPivot() + Vector3.new(0, 2, 0) end
    return topCF
end
local function getTreeClassFromModel(treeModel)
    local tc = treeModel:FindFirstChild("TreeClass")
    if tc then return tc.Value end
    local name = treeModel.Name
    for _, suffix in ipairs({"SpookyNeon","Spooky","GoldSwampy","Cherry","CaveCrawler","MuckySewer"}) do
        if name:find(suffix) then return suffix end
    end
    return "Generic"
end

local function loadWeaponConfigOld(weaponName)
    local module = rep:FindFirstChild("LoadedAssets") and rep.LoadedAssets:FindFirstChild(weaponName.."AxeClass")
    if not module then return nil end
    local success, axeClass = pcall(require, module)
    if not success then return nil end
    local axeInstance = type(axeClass.new)=="function" and (axeClass.new() or axeClass.new({})) or axeClass
    return {Damage=axeInstance.Damage or 10000, SwingCooldown=axeInstance.SwingCooldown or 0.3, SpecialTrees=axeInstance.SpecialTrees or {}}
end

        end

    return bw, bc, bd
end

local axeConfigCacheNew = {}
local function loadWeaponConfigNew(weaponName)
    if axeConfigCacheNew[weaponName] ~= nil then return axeConfigCacheNew[weaponName] end
    if weaponName == "OlReliable" or weaponName == "OIReliable" then
        local override = nil
        local wsTool = workspace:FindFirstChild(lp.Name)
        if wsTool then wsTool = wsTool:FindFirstChild("Tool") end
        if wsTool then
            local tn = wsTool:FindFirstChild("ToolName")
            if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                override = wsTool:FindFirstChild("AxeClassDamageOverride")
            end
        end
        if not override then
            local bp = lp:FindFirstChild("Backpack")
            if bp then
                for _, tool in pairs(bp:GetChildren()) do
                    if tool:IsA("Tool") then
                        local tn = tool:FindFirstChild("ToolName")
                        if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                            override = tool:FindFirstChild("AxeClassDamageOverride")
                            if override then break end
                        end
                    end
                end
            end
        end
        if override then
            local cfg = {Damage=nil, SwingCooldown=nil, Range=nil, SpecialTrees={}}
            for k, v in pairs(override:GetAttributes()) do
                if k == "Damage" then cfg.Damage = v
                elseif k == "SwingCooldown" then cfg.SwingCooldown = v
                elseif k == "Range" then cfg.Range = v
                else cfg.SpecialTrees[k] = {Damage = v} end
            end
            for _, c in pairs(override:GetChildren()) do
                if c:IsA("NumberValue") or c:IsA("IntValue") then
                    if c.Name == "Damage" then cfg.Damage = c.Value
                    elseif c.Name == "SwingCooldown" then cfg.SwingCooldown = c.Value
                    elseif c.Name == "Range" then cfg.Range = c.Value
                    else cfg.SpecialTrees[c.Name] = {Damage = c.Value} end
                end
            end
            if cfg.Damage or cfg.SwingCooldown or next(cfg.SpecialTrees) then
                axeConfigCacheNew[weaponName] = cfg; return cfg
            end
        end
    end
    local function findAxeModule(name)
        local want = name.."AxeClass"
        local la = rep:FindFirstChild("LoadedAssets")
        if la then local m = la:FindFirstChild(want); if m then return m end end
        if type(getloadedmodules) == "function" then
            for _, m in ipairs(getloadedmodules()) do
                if m:IsA("ModuleScript") and (m.Name == want or m:GetFullName():sub(-#want) == want) then return m end
            end
        end
        return nil
    end
    local function decompileSafe(s)
        if not s then return nil end
        if type(decompile) == "function" then
            local ok, src = pcall(decompile, s)
            if ok and type(src) == "string" and #src > 0 then return src end
        end
        return nil
    end
    local function parseNum(src, key)
        local v = src:match("[%w_%.]+%.?" .. key .. "%s*=%s*([%-]?[%d%.]+)")
        if not v then v = src:match(key .. "%s*=%s*([%-]?[%d%.]+)") end
        return tonumber(v)
    end
    local function parseSpecialTrees(src)
        local out = {}
        for tn, body in src:gmatch("SpecialTrees%.([%w_]+)%s*=%s*{(.-)}") do
            local cfg = {}
            cfg.Damage = tonumber(body:match("Damage%s*=%s*([%-]?[%d%.]+)"))
            cfg.SwingCooldown = tonumber(body:match("SwingCooldown%s*=%s*([%-]?[%d%.]+)"))
            if cfg.Damage then out[tn] = cfg end
        end
        return out
    end
    local module = findAxeModule(weaponName)
    if not module then axeConfigCacheNew[weaponName] = nil; return nil end
    local src = decompileSafe(module)
    if not src then axeConfigCacheNew[weaponName] = nil; return nil end
    local cfg = {Damage=parseNum(src,"Damage"), SwingCooldown=parseNum(src,"SwingCooldown"), Range=parseNum(src,"Range"), SpecialTrees=parseSpecialTrees(src)}
    if not cfg.Damage or not cfg.SwingCooldown then axeConfigCacheNew[weaponName] = nil; return nil end
    cfg.Range = cfg.Range or 10
    axeConfigCacheNew[weaponName] = cfg
    return cfg
end
local function selectBestWeaponNew(treeClass)
    local tf = lp:FindFirstChild("ToolFolder"); if not tf then return nil, nil, "未找到 ToolFolder" end

    for _, c in pairs(tf:GetChildren()) do
        local cfg = loadWeaponConfigNew(c.Name)
        if cfg then
            local d = (cfg.SpecialTrees[treeClass] and cfg.SpecialTrees[treeClass].Damage) or cfg.Damage
            if d > bd then bw, bd, bc = c.Name, d, cfg end
        end
    end
    return bw, bc, bd
end

local DANGER_MIN_X, DANGER_MAX_X = -543, -27.8
local DANGER_MIN_Y, DANGER_MAX_Y = -196.7, -148.2
local DANGER_MIN_Z, DANGER_MAX_Z = -2961.7, -2668.7
local function isInDangerZone(pos)
    return pos.X >= DANGER_MIN_X and pos.X <= DANGER_MAX_X
        and pos.Y >= DANGER_MIN_Y and pos.Y <= DANGER_MAX_Y
        and pos.Z >= DANGER_MIN_Z and pos.Z <= DANGER_MAX_Z
end

local function bringTreeOld(treeClass, progressText)
    local weaponName, weaponConfig, damage = selectBestWeaponOld(treeClass)
    if not weaponName then notify("誓言", "自动选武器失败", 3); return false end
    local req = {LoveCave=10000000, Shine=80000000, Magma=5000000, Ice=8000000, Radioactive=10000000, Scale=40000000, Void=5000000, Flame=10000000, BlueFlame=3500000, RainbowFlame=3500000, Celestial=10000000}
    if req[treeClass] and damage < req[treeClass] then notify("誓言", string.format("伤害不足！需要至少 %.0f", req[treeClass]), 4); return false end
    local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown
    local woodSec, tree = findUncutTree(treeClass)
    if not woodSec then notify("誓言", "附近没有可砍的树", 3); return false end
    if isInDangerZone(woodSec.Position) then
        notify("誓言", "树在危险区域内，已删除", 5)
        if tree and tree.Parent then tree:Destroy() end
        return false
    end
    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local tf = lp:FindFirstChild("ToolFolder")
        if tf then local wo = tf:FindFirstChild(weaponName); tool = wo and (wo:IsA("Tool") and wo or wo.Value) or nil end
    end
    if not tool then notify("誓言", "未找到工具: "..weaponName, 3); return false end
    local cutEvent = tree:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无法获取CutEvent", 3); return false end
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then return false end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    lockPlayerAt(getTreeTopCFrame(tree))

local function selectSawmill()
    notify("誓言", "请点击锯木机任意部件", 3)
    local conn
    conn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then return end
        local function isSawmill(root)
            local attr = root:GetAttribute("ItemName")
            if attr and type(attr) == "string" and attr:lower():find("saw") then return true end
            for _, c in pairs(root:GetDescendants()) do
                if c.Name == "ItemName" then
                    local val = c.Value
                    if val and type(val) == "string" and val:lower():find("saw") then return true end
                end
            end
            if root.Name and root.Name:lower():find("saw") then return true end
            local tv = root:FindFirstChild("Type")
            return tv and tv.Value == "Sawmill"
        end
        if not isSawmill(model) then notify("誓言", "不是锯木机", 3); return end
        local function findPart(root)
            for _, c in pairs(root:GetDescendants()) do if c.Name == "Conveyor" and c:IsA("BasePart") then return c, "Conveyor" end end
            for _, c in pairs(root:GetDescendants()) do if c.Name == "Main" and c:IsA("BasePart") then return c, "Main" end end
        end
        local part, pt = findPart(model)
        if not part then notify("誓言", "找不到部件", 3); return end
        bai.cachedSawmill = model; bai.cachedSawmillPos = part.CFrame
        conn:Disconnect()
        notify("誓言", "✅ 已记忆锯木机（"..pt.."）", 3)
    end)
end

local function decomposeAndProcess(treeModel, sawmill, sawmillPos)
    if not sawmillPos then notify("誓言", "锯木机位置无效", 3); return end
    local sections, rootSection = {}, nil
    for _, child in pairs(treeModel:GetChildren()) do
        if child.Name == "WoodSection" and child:IsA("BasePart") and child:FindFirstChild("ID") then
            local id = child.ID.Value
            if id >= 2 then table.insert(sections, {id = id, part = child})
            elseif id == 1 then rootSection = child end
        end
    end
    if #sections == 0 then notify("誓言", "无可分解部分", 3); return end
    table.sort(sections, function(a, b) return a.id > b.id end)
    local cutEvent = treeModel:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无 CutEvent", 3); return end
    local treeClass = getTreeClassFromModel(treeModel)
    local weaponName, weaponConfig, err = selectBestWeaponByPlan(treeClass)
    if not weaponName then notify("誓言", "选武器失败: " .. (err or ""), 3); return end
    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local toolFolder = lp:FindFirstChild("ToolFolder")
        if toolFolder then
            local weaponObj = toolFolder:FindFirstChild(weaponName)
            tool = weaponObj and (weaponObj:IsA("Tool") and weaponObj or weaponObj.Value) or nil
        end
    end
    if not tool then notify("誓言", "未找到工具: " .. weaponName, 3); return end
    local damage = weaponConfig and weaponConfig.Damage or 10000
    local cooldown = weaponConfig and weaponConfig.SwingCooldown or 0.3
    if weaponConfig and weaponConfig.SpecialTrees and weaponConfig.SpecialTrees[treeClass] then
        damage = weaponConfig.SpecialTrees[treeClass].Damage or damage
        cooldown = weaponConfig.SpecialTrees[treeClass].SwingCooldown or cooldown
    end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local initialLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do initialLogs[log] = true end
    local tasks = {}
    notify("誓言", string.format("开始分解 %d 个部分", #sections), 2)
    for _, sec in ipairs(sections) do
        local part = sec.part
        local sectionId = sec.id
        local taskId = task.spawn(function()
            while part and part.Parent do
                pcall(remote.FireServer, remote, cutEvent, {height=0.11, faceVector=Vector3.new(-1,0,0), cooldown=cooldown, sectionId=sectionId, hitPoints=damage, tool=tool, cuttingClass="Axe"})
                task.wait(0.01)
            end
        end)
        tasks[taskId] = true
    end
    local function hasHighIDSections()
        for _, child in pairs(treeModel:GetChildren()) do
            if child.Name == "WoodSection" and child:FindFirstChild("ID") then
                if child.ID.Value >= 2 then return true end
            end
        end
        return false
    end
    local startTime = tick()
    while hasHighIDSections() and tick() - startTime < 10 do task.wait(0.01) end
    for taskId in pairs(tasks) do task.cancel(taskId) end
    if hasHighIDSections() then notify("誓言", "砍树超时", 3); return end
    local allLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do
        if not initialLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp then table.insert(allLogs, log) end
    end
    if #allLogs == 0 then notify("誓言", "未找到木头", 3); return end
    local pos = sawmillPos.p + sawmillPos:VectorToWorldSpace(Vector3.new(0.7, 0, 0))
    local targetPos = CFrame.new(pos) * (sawmillPos - sawmillPos.p) * CFrame.Angles(math.rad(-90), math.rad(90), 0)
    local allModels = {}
    for _, log in ipairs(allLogs) do if log and log.Parent then table.insert(allModels, log) end end
    if rootSection and rootSection.Parent then
        if not treeModel.PrimaryPart then treeModel.PrimaryPart = rootSection end
        table.insert(allModels, treeModel)
    end
    for _, model in ipairs(allModels) do
        if model and model.Parent then
            local primary = model.PrimaryPart or model:FindFirstChild("WoodSection") or model:FindFirstChild("Main") or model:FindFirstChildWhichIsA("BasePart")
            if primary then
                if not model.PrimaryPart then model.PrimaryPart = primary end
                safeTeleport(primary.CFrame + Vector3.new(0, 2, 0))
                task.wait(0.06)
                dragToPosition(model, targetPos)
            end
        end
    end
    notify("誓言", "木头和树根拖拽完成", 3)
end

    if isDecomposeActive then notify("誓言", "分解选择已激活", 2); return end

local function isSelectable(model)
    if not model or not model:IsA("Model") then return false end
    if model:FindFirstChild("BlueprintWoodClass") then return false end
    if model:FindFirstChild("ButtonRemote_SpawnButton") then return false end
    local name = model.Name or ""
    if name:find("Crate") or name:find("Box") or name:find("Gift") then
        local has = false
        local p1 = model:FindFirstChild("PurchasedBoxItemName")
        if p1 and p1.Value and p1.Value ~= "" and p1.Value ~= " " then has = true end
        local p2 = model:FindFirstChild("BoxItemName")
        if not has and p2 and p2.Value and p2.Value ~= "" and p2.Value ~= " " then has = true end
        local p3 = model:FindFirstChild("Type")
        if not has and p3 and p3.Value and p3.Value ~= "" and p3.Value ~= " " then has = true end
        if not has then return false end
    end
    local target = game.Players:FindFirstChild(teleportTargetPlayer)
    if not target then return false end
    local owner = model:FindFirstChild("Owner")
    return owner and owner.Value == target
end
local function highlightItem(model)
    if selectionBoxes[model] then return end
    local primary = model.PrimaryPart or model:FindFirstChild("Main") or model:FindFirstChild("WoodSection") or model:FindFirstChildWhichIsA("BasePart")
    if not primary then return end
    local box = Instance.new("SelectionBox")
    box.Adornee = primary; box.Color3 = Color3.fromRGB(0,255,0); box.Transparency = 0.5; box.Parent = primary
    selectionBoxes[model] = box
end
local function unhighlightItem(model)
    local box = selectionBoxes[model]
    if box then pcall(function() box:Destroy() end); selectionBoxes[model] = nil end
end
local function clearSelection()
    for m, _ in pairs(selectedItems) do unhighlightItem(m) end
    selectedItems = {}
    notify("誓言", "已清空所有选中", 2)
end
local function groupKeyForItem(m)
    local owner = m:FindFirstChild("Owner")
    if owner and owner.Value ~= nil then return m.Name, owner.Value end
    local t = m:FindFirstChild("Type")
    return m.Name, (t and t.Value) or nil
end
local function selectItem(model)
    if not isSelectable(model) then return end
    if selectedItems[model] then
        unhighlightItem(model); selectedItems[model] = nil
    else
        selectedItems[model] = true; highlightItem(model)
    end
end
local function selectSameNameItems(sourceModel)
    local first = sourceModel or next(selectedItems)
    if not first then return end
    local tn, to = groupKeyForItem(first)
    local count = 0
    local function try(m)
        if not m or not m:IsA("Model") then return end
        if not isSelectable(m) then return end
        local n, o = groupKeyForItem(m)
        if n == tn and o == to and not selectedItems[m] then
            selectedItems[m] = true; highlightItem(m); count = count + 1
        end
    end
    try(sourceModel)
    local pm = workspace:FindFirstChild("PlayerModels")
    if pm then for _, m in pairs(pm:GetDescendants()) do try(m) end end
    local lm = workspace:FindFirstChild("LogModels")
    if lm then for _, m in pairs(lm:GetDescendants()) do try(m) end end
    notify("誓言", string.format("已添加 %d 个同名物品", count), 3)
end
local function cancelSameNameItems(sourceModel)
    local tn, to = groupKeyForItem(sourceModel)
    local n = 0
    for m, _ in pairs(selectedItems) do
        local mn, mo = groupKeyForItem(m)
        if mn == tn and mo == to then unhighlightItem(m); selectedItems[m] = nil; n = n + 1 end
    end
    notify("誓言", "已取消本组 " .. n .. " 个", 2)
end
local function teleportSelectedItems()
    if not teleportTargetCF then notify("誓言", "请先设置传送点", 3); return end
    local all = {}
    for m, _ in pairs(selectedItems) do
        if m and m.Parent then
            local primary = m.PrimaryPart or m:FindFirstChild("Main") or m:FindFirstChild("WoodSection") or m:FindFirstChildWhichIsA("BasePart")
            if primary then
                if not m.PrimaryPart then m.PrimaryPart = primary end
                table.insert(all, m)
            end
        end
    end
    if #all == 0 then notify("誓言", "没有选中的物品", 3); return end
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    local originalPos = hrp and hrp.CFrame
    local successCount = 0
    for _, m in ipairs(all) do
        if m.Parent and m.PrimaryPart then
            pcall(function() safeTeleport(m.PrimaryPart.CFrame + Vector3.new(0,2,0)) end)
            task.wait(0.1)
            dragToPosition(m, teleportTargetCF)
            unhighlightItem(m); selectedItems[m] = nil
            successCount = successCount + 1
        end
    end
    if originalPos then safeTeleport(originalPos) end
    notify("誓言", string.format("传送完成 %d/%d", successCount, #all), 3)
end

-- 拖动（顶部 50px）
do
    local dragging, actualDragging = false, false
    local dragStart, startPos = nil, nil
    local DRAG_THRESHOLD = 3
    UIS.InputBegan:Connect(function(input, proc)
        if proc then return end
        if not Main.Visible then return end
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local mx, my = input.Position.X, input.Position.Y
        local mp = Main.AbsolutePosition
        local ms = Main.AbsoluteSize
        if not (mx >= mp.X and mx <= mp.X + ms.X and my >= mp.Y and my <= mp.Y + 50) then return end
        local bhPos = btnHolder.AbsolutePosition
        local bhSize = btnHolder.AbsoluteSize
        if mx >= bhPos.X and mx <= bhPos.X + bhSize.X and my >= bhPos.Y and my <= bhPos.Y + bhSize.Y then return end
        dragging = true; actualDragging = false
        dragStart = input.Position; startPos = Main.Position
    end)
    UIS.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            if not actualDragging then
                if math.abs(delta.X) + math.abs(delta.Y) > DRAG_THRESHOLD then actualDragging = true end
            end
            if actualDragging then
                Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false; actualDragging = false
        end
    end)
end

UIS.InputBegan:Connect(function(input, proc)
    if proc then return end
    if input.KeyCode == Enum.KeyCode.RightShift then OathGui.Enabled = not OathGui.Enabled end
end)

local function makeDropdown(text, order, options, default, onSelect, searchable, parent)

    local SEARCH_H = 34

    crn(listFrame, 8)
    local searchBox = nil
    if searchable then
        local searchBg = new("Frame", {Size = UDim2.new(1, -8, 0, SEARCH_H - 6), Position = UDim2.new(0, 4, 0, 4), BackgroundColor3 = Color3.fromRGB(24, 25, 29), BorderSizePixel = 0, ZIndex = 15}, listFrame)

        searchBox = new("TextBox", {Size = UDim2.new(1, -34, 1, 0), Position = UDim2.new(0, 28, 0, 0), BackgroundTransparency = 1, Text = "", PlaceholderText = "搜索...", PlaceholderColor3 = T.Sub, Font = Enum.Font.Gotham, TextSize = 12, TextColor3 = T.Text, TextXAlignment = Enum.TextXAlignment.Left, ClearTextOnFocus = false, ZIndex = 16}, searchBg)
    end
    local scrollOffsetY = searchable and SEARCH_H or 4
    local scrollFrame = new("ScrollingFrame",{Position = UDim2.new(0, 4, 0, scrollOffsetY), Size = UDim2.new(1, -8, 0, 0), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3, ScrollBarImageColor3 = T.Stroke, CanvasSize = UDim2.new(0, 0, 0, 0), AutomaticCanvasSize = Enum.AutomaticSize.Y, ZIndex = 11}, listFrame)
    new("UIListLayout",{ Padding=UDim.new(0, 2), SortOrder=Enum.SortOrder.LayoutOrder }, scrollFrame)
    local expanded = false
    local expand
    local currentFilter = ""
    local function buildItems()
        for _, ch in ipairs(scrollFrame:GetChildren()) do if ch:IsA("TextButton") then ch:Destroy() end end
        local count = 0
        local lowerFilter = currentFilter:lower()
        for _, opt in ipairs(options) do
            local optStr = tostring(opt)
            local isPlaceholder = (optStr == "--- 请选择树木 ---")
            local matched = (currentFilter == "") or isPlaceholder or optStr:lower():find(lowerFilter, 1, true)
            if matched then
                count = count + 1
                local b = new("TextButton",{Size=UDim2.new(1, 0, 0, ITEM_H - 2), BackgroundTransparency=1, Text="   "..optStr, Font=Enum.Font.Gotham, TextSize=13, TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left, LayoutOrder=count, AutoButtonColor=false, ZIndex=12}, scrollFrame)
                b.MouseButton1Click:Connect(function()

                    if onSelect then pcall(onSelect, cur) end
                    expand(false)
                end)
            end
        end
        return count
    end
    local function updateLayout()
        local cnt = buildItems()
        local visibleListH = math.min(cnt * ITEM_H, 130)
        if cnt == 0 then visibleListH = 30 end
        local totalListH = 8 + (searchable and SEARCH_H or 0) + visibleListH
        TS:Create(listFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, totalListH) }):Play()
        TS:Create(scrollFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, -8, 0, visibleListH) }):Play()

    end

        expanded = state
        if state then
            container.ZIndex = 100
            plusBtn.Text = "−"; listFrame.Visible = true
            if searchBox then searchBox.Text = ""; currentFilter = "" end
            updateLayout()
        else
            container.ZIndex = 1; plusBtn.Text = "+"
            TS:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 0) }):Play()
            TS:Create(container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, HEADER_H) }):Play()
            task.delay(0.2, function() listFrame.Visible = false end)
        end
    end
    if searchBox then
        searchBox:GetPropertyChangedSignal("Text"):Connect(function()
            currentFilter = searchBox.Text or ""
            if expanded then updateLayout() end
        end)
    end
    plusBtn.MouseButton1Click:Connect(function() expand(not expanded) end)

    -- 让外部能刷新选项列表
    local function setOptions(newOptions, newDefault)
        options = newOptions
        if newDefault then
            cur = newDefault
            valueLbl.Text = tostring(cur)
        end
        if expanded then
            updateLayout()
        end
    end

    return container, expand, setOptions
end

-- ==================== 卡密验证系统 ====================
local AUTH_URL = "http://qinscript.lol/api/card/verify"
local CARD_FILE = "DexPlusPlusSetting.json"

local function urlenc(s)
    local ok, enc = pcall(function() return game:GetService("HttpService"):UrlEncode(tostring(s)) end)
    if ok and enc then return enc end
    return tostring(s)
end

local function getHwid()
    local ok, cid = pcall(function()
        return game:GetService("RbxAnalyticsService"):GetClientId()
    end)
    if ok and cid and cid ~= "" then return tostring(cid) end
    return "FALLBACK_" .. tostring(lp.UserId or 0)
end

local function verifyCard(key, hwid, username)
    local keyStr = tostring(key or "")
    local url = AUTH_URL .. "?key=" .. urlenc(keyStr)
        .. "&hwid=" .. urlenc(tostring(hwid))
        .. "&username=" .. urlenc(username or "")
    local ok, result = pcall(function() return game:HttpGet(url) end)
    if not ok or not result then return false, "网络错误" end
    if string.find(result, "OK", 1, true) then return true, "卡密有效" end
    return false, (result ~= "" and result or "卡密无效")
end

local function loadSavedCardKey()
    if writefile and readfile then
        local ok, data = pcall(readfile, CARD_FILE)
        if ok and data then
            data = string.gsub(tostring(data), "%s", "")
            if data ~= "" then return data end
        end
    end
    return nil
end

local function saveCardKey(key)
    if writefile then pcall(writefile, CARD_FILE, tostring(key)) end
end

local stopPurchaseFlag = false

local function waitForItemRefresh(shopItems, itemName, timeout)
    timeout = timeout or 10
    local startTime = tick()
    while tick() - startTime < timeout do
        if stopPurchaseFlag then return false end
        if shopItems:FindFirstChild(itemName) then return true end
        task.wait(0.05)
    end
    return false
end

local function findItemByCD(shopItems, cdValue)
    for _, child in pairs(shopItems:GetChildren()) do
        local settings = child:FindFirstChild("Settings")
        if settings then
            local cd = settings:FindFirstChild("CD")
            if cd and cd:IsA("StringValue") and cd.Value == cdValue then
                return child
            end
        end
    end
    return nil
end

local function waitForItemRefreshByCD(shopItems, cdValue, timeout)
    timeout = timeout or 10
    local startTime = tick()
    while tick() - startTime < timeout do
        if stopPurchaseFlag then return false end
        if findItemByCD(shopItems, cdValue) then return true end
        task.wait(0.05)
    end
    return false
end

local function purchaseCycle(storeInfo, selectedItemName, buyAmount, doubleConfirm, itemCD)
    local store = storeInfo.object
    local shopItems = storeInfo.shopItems
    local npcInfo = storeInfo.npc
    if not npcInfo or not npcInfo.id then
        notify("誓言", "商店 "..storeInfo.name.." NPC ID 无效", 5)
        return false
    end
    local npcChar = store:FindFirstChild(npcInfo.path)
    if not npcChar then notify("誓言", "找不到NPC: "..npcInfo.path, 3); return false end
    local counter = store:FindFirstChild("Counter")
    if not counter or not counter:IsA("BasePart") then notify("誓言", "商店 "..storeInfo.name.." 没有Counter", 3); return false end
    local NPCDialog = REPLICATED_STORAGE:WaitForChild("NPCDialog")

    local function sendPurchaseAndCollect()
        local purchasedItems = {}
        local conn = workspace.PlayerModels.ChildAdded:Connect(function(model)
            if model:IsA("Model") and model:FindFirstChild("Owner") and model.Owner.Value == lp then
                table.insert(purchasedItems, model)
            end
        end)
        local args = {{ID=npcInfo.id, Character=npcChar, Name=npcInfo.name}, "ConfirmPurchase"}
        local ok, err = pcall(function()
            NPCDialog.PlayerChatted:InvokeServer(table.unpack(args))
        end)
        if not ok then
            notify("誓言", "购买失败: "..tostring(err), 3)
            conn:Disconnect()
            return nil
        end
        task.wait(0.04)
        conn:Disconnect()
        return purchasedItems
    end

    local originalPos = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not originalPos then notify("誓言", "无法获取玩家位置", 3); return false end
    originalPos = originalPos.CFrame

    local successCycles = 0
    for i = 1, buyAmount do
        if stopPurchaseFlag then break end
        local itemModel
        if itemCD then
            itemModel = findItemByCD(shopItems, itemCD)
            if not itemModel then
                if not waitForItemRefreshByCD(shopItems, itemCD, 15) then
                    if stopPurchaseFlag then break end
                    notify("誓言", "等待 CD 刷新超时", 3)
                    break
                end
                itemModel = findItemByCD(shopItems, itemCD)
                if not itemModel then break end
            end
        else
            itemModel = shopItems:FindFirstChild(selectedItemName)
            if not itemModel then
                if not waitForItemRefresh(shopItems, selectedItemName, 15) then
                    if stopPurchaseFlag then break end
                    notify("誓言", "等待商品刷新超时", 3)
                    break
                end
                itemModel = shopItems:FindFirstChild(selectedItemName)
                if not itemModel then break end
            end
        end

        local itemPrimary = itemModel.PrimaryPart or itemModel:FindFirstChild("WoodSection") or itemModel:FindFirstChild("Main") or itemModel:FindFirstChildWhichIsA("BasePart")
        if not itemPrimary then notify("誓言", "无法定位商品模型", 3); break end
        if not itemModel.PrimaryPart then itemModel.PrimaryPart = itemPrimary end
        if stopPurchaseFlag then break end

        lockPlayerAt(itemPrimary.CFrame + Vector3.new(0,3,0))
        task.wait(0.2)
        if stopPurchaseFlag then break end

        local counterSurface = counter.CFrame + Vector3.new(0,0.6,0)
        dragToPosition(itemModel, counterSurface)
        if stopPurchaseFlag then break end

        lockPlayerAt(counter.CFrame + Vector3.new(0,3,0))
        task.wait(0.1)

        local allItems = {}
        local requestCount = doubleConfirm and 2 or 2
        for k = 1, requestCount do
            if stopPurchaseFlag then break end
            local items = sendPurchaseAndCollect()
            if items then
                for _, it in ipairs(items) do table.insert(allItems, it) end
            else
                break
            end
            if k < requestCount then task.wait(0.01) end
        end

        for _, item in ipairs(allItems) do
            if stopPurchaseFlag then break end
            if not item or not item.Parent then continue end
            if not item.PrimaryPart then
                local part = item:FindFirstChild("Main") or item:FindFirstChild("WoodSection") or item:FindFirstChildWhichIsA("BasePart")
                if part then item.PrimaryPart = part end
            end
            dragToPosition(item, originalPos)
            task.wait(0.05)
        end
        successCycles = successCycles + 1

        if i < buyAmount and not stopPurchaseFlag then
            local refreshed
            if itemCD then
                refreshed = waitForItemRefreshByCD(shopItems, itemCD, 30)
            else
                refreshed = waitForItemRefresh(shopItems, selectedItemName, 30)
            end
            if not refreshed then
                if stopPurchaseFlag then break end
                notify("誓言", "等待商品刷新超时，停止后续循环", 3)
                break
            end
        end
    end

    unlockPlayer()
    safeTeleport(originalPos)
    if stopPurchaseFlag then
        notify("誓言", "购买已中断，已返回原地", 3)
        stopPurchaseFlag = false
        return false
    else
        notify("誓言", storeInfo.name..string.format(" 完成 %d / %d 次", successCycles, buyAmount), 4)
        return successCycles == buyAmount
    end
end

-- ==================== 一键购买所有蓝图 ====================
local function buyAllBlueprints()
    stopPurchaseFlag = false
    local startHRP = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    local startCF = startHRP and startHRP.CFrame
    local blueprintFolder = lp:FindFirstChild("PlayerBlueprints") and lp.PlayerBlueprints:FindFirstChild("Blueprints")
    local targets, seen = {}, {}
    for _, store in ipairs(allStores) do
        if store.shopItems then
            for _, item in pairs(store.shopItems:GetChildren()) do
                local typeVal = item:FindFirstChild("Type")
                local tstr = (typeVal and typeVal.Value ~= nil) and tostring(typeVal.Value) or ""
                if tstr == "Blueprint" then
                    local nm = tostring(item.Name)
                    if nm ~= "" and not seen[nm] then
                        seen[nm] = true
                        if not (blueprintFolder and blueprintFolder:FindFirstChild(nm)) then
                            targets[#targets+1] = nm
                        end
                    end
                end
            end
        end
    end
    table.sort(targets)
    if #targets == 0 then
        notify("誓言", "已拥有全部蓝图，或没找到可购买蓝图", 3)
        return
    end
    notify("誓言", string.format("还缺 %d 种蓝图，开始购买", #targets), 4)
    local bought, stillMissing = 0, {}
    for _, nm in ipairs(targets) do
        if stopPurchaseFlag then break end
        local box, err = _G.qSnowBuyBlueprintBox(nm)
        if not box then
            table.insert(stillMissing, nm)
            notify("誓言", "买不到 "..nm.."："..tostring(err), 3)
        else
            openBlueprintBox(box)
            local waited = 0
            while waited < 5 and not (blueprintFolder and blueprintFolder:FindFirstChild(nm)) do
                task.wait(0.3)
                waited = waited + 0.3
                if stopPurchaseFlag then break end
            end
            if blueprintFolder and blueprintFolder:FindFirstChild(nm) then
                bought = bought + 1
                notify("誓言", "已得到蓝图: "..nm, 2)
            else
                table.insert(stillMissing, nm)
            end
        end
        task.wait(0.2)
    end
    if startCF then safeTeleport(startCF) end
    if stopPurchaseFlag then
        notify("誓言", "购买已停止，已返回原地", 3)
        return
    end
    if #stillMissing > 0 then
        local txt = table.concat(stillMissing, "\n")
        pcall(function() setclipboard(txt) end)
        notify("誓言", string.format("本次新增 %d 种，仍缺 %d 种（已复制到剪贴板）", bought, #stillMissing), 8)
    else
        notify("誓言", string.format("购买完成，本次新增 %d 种蓝图", bought), 5)
    end
end

-- ==================== 全局购买函数（供蓝图工具调用） ====================
_G.qSnowBuyBlueprintBox = function(blueprintName)
    if stopPurchaseFlag then return nil,"已停止" end
    local targetStore, targetItemName = nil, nil
    local lower = string.lower(tostring(blueprintName))
    for _, s in ipairs(allStores) do
        if s.shopItems then
            for _, item in pairs(s.shopItems:GetChildren()) do
                if string.lower(tostring(item.Name)) == lower then
                    targetStore, targetItemName = s, item.Name
                    break
                end
            end
        end
        if targetStore then break end
    end
    if not targetStore then return nil,"没有任何商店出售蓝图: "..tostring(blueprintName) end
    if not targetStore.npc or not targetStore.npc.id then return nil,"商店 "..targetStore.name.." 的 NPC ID 未获取" end

    local store = targetStore.object
    local npcInfo = targetStore.npc
    local npcChar = store:FindFirstChild(npcInfo.path)
    if not npcChar then return nil,"找不到NPC: "..tostring(npcInfo.path) end
    local counter = store:FindFirstChild("Counter")
    if not counter or not counter:IsA("BasePart") then return nil,"商店无 Counter" end
    local shopItems = store:FindFirstChild("ShopItems")
    if not shopItems then return nil,"无 ShopItems" end

    local itemModel = shopItems:FindFirstChild(targetItemName)
    if not itemModel then return nil,"商品不在货架: "..targetItemName end
    local primary = itemModel.PrimaryPart or itemModel:FindFirstChild("WoodSection") or itemModel:FindFirstChild("Main") or itemModel:FindFirstChildWhichIsA("BasePart")
    if not primary then return nil,"无法定位商品模型" end
    if not itemModel.PrimaryPart then itemModel.PrimaryPart = primary end

    lockPlayerAt(primary.CFrame + Vector3.new(0,3,0))
    task.wait(0.3)
    dragToPosition(itemModel, counter.CFrame + Vector3.new(0,0.6,0))
    task.wait(0.09)
    lockPlayerAt(counter.CFrame + Vector3.new(0,3,0))
    task.wait(0.1)

    local purchased = {}
    local conn = workspace.PlayerModels.ChildAdded:Connect(function(m)
        if m:IsA("Model") and m:FindFirstChild("Owner") and m.Owner.Value == lp then
            table.insert(purchased, m)
        end
    end)
    local NPCDialog = REPLICATED_STORAGE:WaitForChild("NPCDialog")
    local args = {{ID=npcInfo.id, Character=npcChar, Name=npcInfo.name}, "ConfirmPurchase"}
    pcall(function() NPCDialog.PlayerChatted:InvokeServer(table.unpack(args)) end)
    task.wait(0.1)
    pcall(function() NPCDialog.PlayerChatted:InvokeServer(table.unpack(args)) end)
    task.wait(0.1)
    conn:Disconnect()
    unlockPlayer()
    for i = #purchased, 1, -1 do
        local m = purchased[i]
        if m and m.Parent then return m, targetStore.name end
    end
    return nil,"购买后未找到新盒子"
end

-- ==================== 卡密验证系统 ====================
local AUTH_URL = "http://qinscript.lol/api/card/verify"
local CARD_FILE = "DexPlusPlusSetting.json"

local function urlenc(s)
    local ok, enc = pcall(function() return game:GetService("HttpService"):UrlEncode(tostring(s)) end)
    if ok and enc then return enc end
    return tostring(s)
end

local function getHwid()
    local ok, cid = pcall(function()
        return game:GetService("RbxAnalyticsService"):GetClientId()
    end)
    if ok and cid and cid ~= "" then return tostring(cid) end
    return "FALLBACK_" .. tostring(lp.UserId or 0)
end

local function verifyCard(key, hwid, username)
    local keyStr = tostring(key or "")
    local url = AUTH_URL .. "?key=" .. urlenc(keyStr)
        .. "&hwid=" .. urlenc(tostring(hwid))
        .. "&username=" .. urlenc(username or "")
    local ok, result = pcall(function() return game:HttpGet(url) end)
    if not ok or not result then return false, "网络错误" end
    if string.find(result, "OK", 1, true) then return true, "卡密有效" end
    return false, (result ~= "" and result or "卡密无效")
end

local function loadSavedCardKey()
    if writefile and readfile then
        local ok, data = pcall(readfile, CARD_FILE)
        if ok and data then
            data = string.gsub(tostring(data), "%s", "")
            if data ~= "" then return data end
        end
    end
    return nil
end

local function saveCardKey(key)
    if writefile then pcall(writefile, CARD_FILE, tostring(key)) end
end

    pcall(function() npcDialog.SetChattingValue:InvokeServer(0) end)

local function waitForItemRefresh(shopItems, itemName, timeout)
    timeout = timeout or 10

        if stopPurchaseFlag then return false end

local function findItemByCD(shopItems, cdValue)
    for _, child in pairs(shopItems:GetChildren()) do
        local settings = child:FindFirstChild("Settings")
        if settings then
            local cd = settings:FindFirstChild("CD")
            if cd and cd:IsA("StringValue") and cd.Value == cdValue then

end

local function purchaseCycle(storeInfo, selectedItemName, buyAmount, doubleConfirm, itemCD)
    local store = storeInfo.object
    local shopItems = storeInfo.shopItems
    local npcInfo = storeInfo.npc
    if not npcInfo or not npcInfo.id then
        notify("誓言", "商店 "..storeInfo.name.." NPC ID 无效", 5)
        return false
    end
    local npcChar = store:FindFirstChild(npcInfo.path)
    if not npcChar then notify("誓言", "找不到NPC: "..npcInfo.path, 3); return false end
    local counter = store:FindFirstChild("Counter")
    if not counter or not counter:IsA("BasePart") then notify("誓言", "商店 "..storeInfo.name.." 没有Counter", 3); return false end
    local NPCDialog = REPLICATED_STORAGE:WaitForChild("NPCDialog", 5)
    if not NPCDialog then notify("誓言", "NPCDialog 不存在", 3); return false end

    local function sendPurchaseAndCollect()
        local purchasedItems = {}
        local conn = workspace.PlayerModels.ChildAdded:Connect(function(model)
            if model:IsA("Model") and model:FindFirstChild("Owner") and model.Owner.Value == lp then
                table.insert(purchasedItems, model)
            end
        end)
        local args = {{ID=npcInfo.id, Character=npcChar, Name=npcInfo.name}, "ConfirmPurchase"}
        local ok, err = pcall(function()
            NPCDialog.PlayerChatted:InvokeServer(table.unpack(args))
        end)
        if not ok then
            notify("誓言", "购买失败: "..tostring(err), 3)
            conn:Disconnect()
            return nil
        end
        task.wait(0.04)
        conn:Disconnect()
        return purchasedItems
    end

    local originalPos = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if not originalPos then notify("誓言", "无法获取玩家位置", 3); return false end
    originalPos = originalPos.CFrame

    local successCycles = 0
    for i = 1, buyAmount do
        if stopPurchaseFlag then break end
        local itemModel
        if itemCD then
            itemModel = findItemByCD(shopItems, itemCD)
            if not itemModel then
                if not waitForItemRefreshByCD(shopItems, itemCD, 15) then
                    if stopPurchaseFlag then break end
                    notify("誓言", "等待 CD 刷新超时", 3)
                    break
                end
                itemModel = findItemByCD(shopItems, itemCD)
                if not itemModel then break end
            end
        else
            itemModel = shopItems:FindFirstChild(selectedItemName)
            if not itemModel then
                if not waitForItemRefresh(shopItems, selectedItemName, 15) then
                    if stopPurchaseFlag then break end
                    notify("誓言", "等待商品刷新超时", 3)
                    break
                end
                itemModel = shopItems:FindFirstChild(selectedItemName)
                if not itemModel then break end
            end
        end

        local itemPrimary = itemModel.PrimaryPart or itemModel:FindFirstChild("WoodSection") or itemModel:FindFirstChild("Main") or itemModel:FindFirstChildWhichIsA("BasePart")
        if not itemPrimary then notify("誓言", "无法定位商品模型", 3); break end
        if not itemModel.PrimaryPart then itemModel.PrimaryPart = itemPrimary end
        if stopPurchaseFlag then break end

        lockPlayerAt(itemPrimary.CFrame + Vector3.new(0,3,0))
        task.wait(0.2)
        if stopPurchaseFlag then break end

        local counterSurface = counter.CFrame + Vector3.new(0,0.6,0)
        dragToPosition(itemModel, counterSurface)
        if stopPurchaseFlag then break end

        lockPlayerAt(counter.CFrame + Vector3.new(0,3,0))
        task.wait(0.1)

        local allItems = {}
        local requestCount = 2
        for k = 1, requestCount do
            if stopPurchaseFlag then break end
            local items = sendPurchaseAndCollect()
            if items then
                for _, it in ipairs(items) do table.insert(allItems, it) end
            else
                break
            end
            if k < requestCount then task.wait(0.01) end
        end

        for _, item in ipairs(allItems) do
            if stopPurchaseFlag then break end
            if not item or not item.Parent then continue end
            if not item.PrimaryPart then
                local part = item:FindFirstChild("Main") or item:FindFirstChild("WoodSection") or item:FindFirstChildWhichIsA("BasePart")
                if part then item.PrimaryPart = part end
            end
            dragToPosition(item, originalPos)
            task.wait(0.05)
        end
        successCycles = successCycles + 1

        if i < buyAmount and not stopPurchaseFlag then
            local refreshed
            if itemCD then
                refreshed = waitForItemRefreshByCD(shopItems, itemCD, 30)
            else
                refreshed = waitForItemRefresh(shopItems, selectedItemName, 30)
            end
            if not refreshed then
                if stopPurchaseFlag then break end
                notify("誓言", "等待商品刷新超时，停止后续循环", 3)
                break
            end
        end
    end

    unlockPlayer()
    safeTeleport(originalPos)
    if stopPurchaseFlag then
        notify("誓言", "购买已中断，已返回原地", 3)
        stopPurchaseFlag = false
        return false
    else
        notify("誓言", storeInfo.name..string.format(" 完成 %d / %d 次", successCycles, buyAmount), 4)
        return successCycles == buyAmount
    end
end

            local txt = table.concat(stillMissing, "\n")
            pcall(function() setclipboard(txt) end)
            notify("誓言", string.format("本次新增 %d 种，仍缺 %d 种", bought, #stillMissing), 8)
        else
            notify("誓言", string.format("购买完成，本次新增 %d 种蓝图", bought), 5)
        end
    end)
end, allItemSec)

local allBuyAmount = 1
makeBox("循环次数", "1", 13, "1", function(v)
    local n = tonumber(v); if n and n > 0 then allBuyAmount = math.floor(n) end
end, allItemSec)

local allDoubleConfirm = false
makeToggle("双重确认", 14, false, function(s) allDoubleConfirm = s end, allItemSec)

makeBtn("开始购买", 15, function()
    if not storesReady then notify("誓言", "商店数据尚未加载完成，请稍候", 3); return end
    if not selectedAllDisplay then notify("誓言", "请选择商品", 3); return end
    local entry = allItemMap[selectedAllDisplay]
    if not entry then notify("誓言", "请选择商品", 3); return end
    local store = entry.store
    if not store.npc or not store.npc.id then
        notify("誓言", "商店 '"..store.name.."' 无法购买（ID无效）", 3)
        return
    end
    stopPurchaseFlag = false
    task.spawn(function()
        pcall(function()
            purchaseCycle(store, entry.item, allBuyAmount, allDoubleConfirm)
        end)
    end)
end, allItemSec)

makeBtn("停止购买", 16, function()
    stopPurchaseFlag = true
    notify("誓言", "已发送停止信号", 3)
end, allItemSec)

-- ==================== 各商店分区容器（先占位） ====================
local storeSecContainer = new("Frame", {
    Size = UDim2.new(1,0,0,0), BackgroundTransparency=1,
    AutomaticSize = Enum.AutomaticSize.Y, LayoutOrder = 20,
}, pBuy)
new("UIListLayout", {Padding=UDim.new(0,6), SortOrder=Enum.SortOrder.LayoutOrder}, storeSecContainer)

-- ==================== 异步：加载商店数据并构建 UI ====================
task.spawn(function()
    -- 等待 Stores 出现（不阻塞主线程）
    local ok, stores = pcall(function()
        return ws:WaitForChild("Stores", 30)
    end)
    if not ok or not stores then
        allItemStatusLabel.Text = "   加载商店失败：Stores 不存在"
        return
    end
    StoresFolder = stores

    -- 获取 NPC ID
    local initialIds = GetAllCashierIds()
    for storeName, info in pairs(storeInfoMapping) do
        local npc = npcMapping[storeName]
        if initialIds[info.name] then npc.id = initialIds[info.name] end
    end

    -- 扫描商店
    for _, store in pairs(StoresFolder:GetChildren()) do
        local shopItems = store:FindFirstChild("ShopItems")
        if shopItems and #shopItems:GetChildren() > 0 then
            local storeName = store.Name
            local npcInfo = npcMapping[storeName]
            if npcInfo then
                local itemsList = {}
                local isMusic = (storeName == "MusicStore")
                for _, item in pairs(shopItems:GetChildren()) do
                    local displayName = item.Name
                    if isMusic then
                        local settings = item:FindFirstChild("Settings")
                        if settings then
                            local cd = settings:FindFirstChild("CD")
                            if cd and cd:IsA("StringValue") and cd.Value ~= "" then
                                displayName = cd.Value
                            end
                        end
                    end
                    table.insert(itemsList, {model=item, display=displayName})
                end
                table.insert(allStores, {
                    name = storeName,
                    object = store,
                    shopItems = shopItems,
                    npc = npcInfo,
                    itemsList = itemsList,
                    isMusicStore = isMusic
                })
            end
        end
    end
    table.sort(allStores, function(a,b) return a.name < b.name end)

    storesReady = true

    -- 更新所有商品下拉框
    for _, store in ipairs(allStores) do
        if store.shopItems then
            for _, item in pairs(store.shopItems:GetChildren()) do
                local display = store.name.." - "..item.Name
                allItemMap[display] = {store = store, item = item.Name}
            end
        end
    end

    local allOptions = {}
    for k, _ in pairs(allItemMap) do table.insert(allOptions, k) end
    table.sort(allOptions)

    if #allOptions == 0 then
        allItemStatusLabel.Text = "   暂无商品"
    else
        allItemStatusLabel.Text = string.format("   共加载 %d 种商品", #allOptions)
        local _, _, setOpt = makeDropdown("选择商品", 11, allOptions, allOptions[1], function(sel)
            selectedAllDisplay = sel
        end, true, allItemSec)
        selectedAllDisplay = allOptions[1]
    end

    -- 为每个商店建立独立分区
    local order = 0
    for _, store in ipairs(allStores) do
        order = order + 1
        local storeName = store.name
        local shopItemsFolder = store.shopItems
        local npcInfo = store.npc
        local section = makeCollapsible(storeName, order, storeSecContainer)

        local statusText = (npcInfo.id and "NPC ID: "..tostring(npcInfo.id).." ✅") or "等待获取"
        local statusLabel = new("TextLabel", {
            Size = UDim2.new(1,0,0,24), BackgroundColor3=T.Panel, BorderSizePixel=0,
            Text = "   "..statusText, Font = Enum.Font.Gotham, TextSize = 12, TextColor3 = T.Sub,
            TextXAlignment = Enum.TextXAlignment.Left, LayoutOrder = 100,
        }, section)
        crn(statusLabel, 6)
        npcInfo.statusLabel = statusLabel

        local rawItems = {}
        for _, item in pairs(shopItemsFolder:GetChildren()) do
            table.insert(rawItems, {name = item.Name, model = item})
        end
        table.sort(rawItems, function(a,b) return a.name < b.name end)

        local displayOptions, valueMap = {}, {}
        local isMusic = (storeName == "MusicStore")
        for _, entry in ipairs(rawItems) do
            local displayName = entry.name
            local identifier = entry.name
            if isMusic then
                local settings = entry.model:FindFirstChild("Settings")
                if settings then
                    local cd = settings:FindFirstChild("CD")
                    if cd and cd:IsA("StringValue") and cd.Value ~= "" then
                        displayName = cd.Value
                        identifier = cd.Value
                    end
                end
            end
            displayOptions[#displayOptions+1] = displayName
            valueMap[displayName] = identifier
        end

        if #displayOptions == 0 then
            new("TextLabel", {Size=UDim2.new(1,0,0,30), BackgroundTransparency=1, Text="该商店暂无商品", Font=Enum.Font.Gotham, TextSize=13, TextColor3=T.Sub, LayoutOrder=101}, section)
        else
            local selectedIdentifier = valueMap[displayOptions[1]]
            makeDropdown("选择商品", 101, displayOptions, displayOptions[1], function(d)
                selectedIdentifier = valueMap[d]
            end, true, section)

            local buyAmount = 1
            makeBox("循环次数", "1", 102, "1", function(v)
                local n = tonumber(v); if n and n > 0 then buyAmount = math.floor(n) end
            end, section)

            local doubleConfirmFlag = false
            makeToggle("双重确认", 103, false, function(s) doubleConfirmFlag = s end, section)

            makeBtn("传送至柜台", 104, function()
                local counter = store.object:FindFirstChild("Counter")
                if counter and counter:IsA("BasePart") then
                    safeTeleport(counter.CFrame + Vector3.new(0,2.5,0))
                    notify("誓言", string.format("已传送到 %s 柜台", storeName), 2)
                else
                    notify("誓言", string.format("商店 %s 没有 Counter", storeName), 3)
                end
            end, section)

            makeBtn("开始循环购买", 105, function()
                if not npcInfo.id then
                    notify("誓言", "商店 "..storeName.." ID 尚未获取", 3)
                    return
                end
                if not selectedIdentifier then notify("誓言", "未选中商品", 3); return end
                stopPurchaseFlag = false
                task.spawn(function()
                    pcall(function()
                        if isMusic then
                            purchaseCycle(store, nil, buyAmount, doubleConfirmFlag, selectedIdentifier)
                        else
                            purchaseCycle(store, selectedIdentifier, buyAmount, doubleConfirmFlag, nil)
                        end
                    end)
                end)
            end, section)

            makeBtn("停止购买", 106, function()
                stopPurchaseFlag = true
                notify("誓言", "已发送停止信号", 3)
            end, section)

            new("TextLabel", {Size=UDim2.new(1,0,0,20), BackgroundTransparency=1, Text="  共 "..#rawItems.." 种商品 · NPC: "..npcInfo.name, Font=Enum.Font.Gotham, TextSize=11, TextColor3=T.Sub, TextXAlignment=Enum.TextXAlignment.Left, LayoutOrder=107}, section)
        end
    end

    -- 后台继续补齐 NPC ID
    task.spawn(function()
        local noProgress = 0
        while true do
            local missingCount = 0
            for _, npc in pairs(npcMapping) do
                if npc.id == nil then missingCount = missingCount + 1 end
            end
            if missingCount == 0 then break end
            task.wait(2)
            local newIds = GetAllCashierIds()
            local gained = 0
            for storeName, info in pairs(storeInfoMapping) do
                local npc = npcMapping[storeName]
                if npc and npc.id == nil and newIds[info.name] then
                    npc.id = newIds[info.name]
                    gained = gained + 1
                    if npc.statusLabel then
                        pcall(function() npc.statusLabel.Text = "   NPC ID: "..tostring(npc.id).." ✅" end)
                    end
                end
            end
            if gained == 0 then
                noProgress = noProgress + 1
                if noProgress >= 3 then break end
            else
                noProgress = 0
            end
        end
    end)
end)

print("[誓言面板] 第3段 已加载")
print("[誓言面板] 全部加载完成，按 右Shift 打开面板")
