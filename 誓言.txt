local function new(c,p,par) local i=Instance.new(c); for k,v in next,p do i[k]=v end; if par then i.Parent=par end; return i end

--  主面板（一开始就建，先隐藏）
-- ============================================================
local OathGui = new("ScreenGui",{Name="OathPanel", ResetOnSpawn=false,
    ZIndexBehavior=Enum.ZIndexBehavior.Sibling, IgnoreGuiInset=true, Enabled=false}, PG)

local Main = new("Frame",{Size=UDim2.fromOffset(620,400), Position=UDim2.fromScale(0.5,0.5),
    AnchorPoint=Vector2.new(0.5,0.5), BackgroundColor3=T.Bg, BorderSizePixel=0}, OathGui)
crn(Main,16)

local Sb = new("Frame",{Size=UDim2.new(0,150,1,0), BackgroundColor3=T.Sidebar, BorderSizePixel=0}, Main)
crn(Sb,16)
new("TextLabel",{Size=UDim2.new(0,130,0,20), Position=UDim2.new(0,14,0,6), BackgroundTransparency=1,
    Text="木材大亨2", Font=Enum.Font.GothamBold, TextSize=16, TextColor3=T.Accent,
    TextXAlignment=Enum.TextXAlignment.Left}, Sb)

local tabs = {"信息","功能","脚本","设置","关于"}
local tabBtns = {}
local pc = new("Frame",{Position=UDim2.new(0,150,0,0), Size=UDim2.new(1,-150,1,0), BackgroundTransparency=1}, Main)
local pages = {}
for i, name in ipairs(tabs) do
    local p = new("ScrollingFrame",{Name=name, Size=UDim2.fromScale(1,1), BackgroundTransparency=1,
        BorderSizePixel=0, ScrollBarThickness=3, CanvasSize=UDim2.new(0,0,0,0),
        AutomaticCanvasSize=Enum.AutomaticSize.Y, Visible=false}, pc)
    new("UIListLayout",{Padding=UDim.new(0,4), SortOrder=Enum.SortOrder.LayoutOrder}, p)
    new("UIPadding",{PaddingTop=UDim.new(0,48), PaddingBottom=UDim.new(0,10),
        PaddingLeft=UDim.new(0,14), PaddingRight=UDim.new(0,14)}, p)
    pages[i] = p
end

local h = new("Frame",{Position=UDim2.new(0,0,0,56), Size=UDim2.new(1,0,1,-64), BackgroundTransparency=1}, Sb)
new("UIListLayout",{Padding=UDim.new(0,4), HorizontalAlignment=Enum.HorizontalAlignment.Center}, h)
for i, name in ipairs(tabs) do
    local b = new("TextButton",{Size=UDim2.new(1,0,0,32), BackgroundTransparency=1, Text="",
        AutoButtonColor=false, LayoutOrder=i}, h)
    crn(b,16)
    local l = new("TextLabel",{Position=UDim2.new(0,14,0,0), Size=UDim2.new(1,-14,1,0),
        BackgroundTransparency=1, Text=name, Font=Enum.Font.GothamMedium, TextSize=13,
        TextColor3=T.Sub, TextXAlignment=Enum.TextXAlignment.Left}, b)
    tabBtns[i] = {button=b, label=l}
    b.MouseButton1Click:Connect(function()
        for j, p in ipairs(pages) do
            p.Visible = (j == i)
            tabBtns[j].label.TextColor3 = (j == i) and T.Text or T.Sub
        end
    end)
end
pages[1].Visible = true
tabBtns[1].label.TextColor3 = T.Text

-- 信息页
local info = pages[1]
local ic = new("Frame",{Size=UDim2.new(1,0,0,80), BackgroundColor3=T.Panel, BorderSizePixel=0}, info)
crn(ic,10)
new("TextLabel",{Size=UDim2.new(1,-20,0,30), Position=UDim2.new(0,10,0,10),
    BackgroundTransparency=1, Text="木材大亨2 · 誓言面板",
    Font=Enum.Font.GothamBold, TextSize=14, TextColor3=T.Text}, ic)
new("TextLabel",{Size=UDim2.new(1,-20,0,20), Position=UDim2.new(0,10,0,40),
    BackgroundTransparency=1, Text="作者：是我太傻",
    Font=Enum.Font.Gotham, TextSize=11, TextColor3=T.Sub}, ic)

-- 功能页占位
local ft = new("TextLabel",{Size=UDim2.new(1,0,0,30), BackgroundTransparency=1,
    Text="功能分区（待添加）", Font=Enum.Font.GothamBold, TextSize=14, TextColor3=T.Sub}, pages[2])

-- 拖动
do
    local dragging, ds, sp
    Main.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=true; ds=i.Position; sp=Main.Position
            i.Changed:Connect(function() if i.UserInputState==Enum.UserInputState.End then dragging=false end end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            local d = i.Position - ds
            Main.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
        end
    end)
end

print("[誓言面板] 主面板已创建")

local verifying = false
local function verify()
    if verifying then return end
    verifying = true
    if KEYS[ib.Text] then
        tip.TextColor3 = T.Accent
        tip.Text = "验证成功，正在加载..."
        print("[誓言面板] 卡密正确，显示主面板")
        OathGui.Enabled = true
        task.wait(0.3)
        keyGui:Destroy()
    else
        tip.Text = "卡密错误，请重试"
        ib.Text = ""
        verifying = false
    end
end
conf.MouseButton1Click:Connect(verify)
ib.FocusLost:Connect(function(e) if e then verify() end end)
