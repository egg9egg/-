local TARGET_GAME_NAMES = {
    "木材大亨", "Lumber", "Lumber Tycoon",
    "最强战场", "Strongest Battlegrounds", "The Strongest Battlegrounds",
    "每秒+1个方块", "每秒加一个方块", "每秒+1方块",
    "每秒 +1 个方块", "每秒 +1个方块", "每秒+1 个方块",
    "1 Block Per Second",
    "下水道",
}
local TARGET_PLACE_IDS = { 13822889 }

local function detectGame()
    for _, id in ipairs(TARGET_PLACE_IDS) do
        if game.PlaceId == id then return "Lumber", "木材大亨" end
    end
    local ok, info = pcall(function()
        return MarketplaceService:GetProductInfo(game.PlaceId)
    end)
    if ok and info and info.Name then
        local n = info.Name
        if string.find(n, "木材大亨", 1, true) or string.find(n, "Lumber", 1, true) then
            return "Lumber", "木材大亨"
        end
        if string.find(n, "最强战场", 1, true)
        or string.find(n, "Strongest Battlegrounds", 1, true) then
            return "Battlegrounds", "最强战场"
        end
        if string.find(n, "下水道", 1, true) or string.find(n, "Sewer", 1, true) then
            return "Sewer", "下水道"
        end
        local noSpace = n:gsub("%s", "")
        if string.find(noSpace, "每秒", 1, true) and string.find(noSpace, "方块", 1, true) then
            return "BlockPerSecond", "每秒+1个方块"
        end
        if string.find(n, "1 Block Per Second", 1, true) then
            return "BlockPerSecond", "每秒+1个方块"
        end
        return "Generic", "通用面板"
    end
    return "Generic", "通用面板"
end

--=========================================================
--  工具函数
--=========================================================
local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in next, props do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function corner(parent, r)
    return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, parent)
end

local function stroke(parent, color, thickness, transparency)
    return new("UIStroke", {
        Color = color or Theme.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

            AnchorPoint = Vector2.new(0, 0.5),
            BackgroundTransparency = 1,
            Text = "📁",
            Font = Enum.Font.GothamBold,
            TextSize = 20,
            TextColor3 = Color3.fromRGB(255, 255, 255),
            TextXAlignment = Enum.TextXAlignment.Center,
            TextYAlignment = Enum.TextYAlignment.Center,
        }, flyHeader)

        new("TextLabel", {
            Position = UDim2.new(0, 38, 0, 0),
            Size = UDim2.new(1, -38, 1, 0),
            BackgroundTransparency = 1,
            Text = "飞行",
            Font = Enum.Font.GothamMedium,
            TextSize = 13,
            TextColor3 = Theme.Text,
            TextXAlignment = Enum.TextXAlignment.Left,
        }, flyHeader)

        local flyHeaderBtn = new("TextButton", {
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            Text = "",
            AutoButtonColor = false,
            ZIndex = 5,
        }, flyHeader)

        local flyContainer = new("Frame", {
            Size = UDim2.new(1, 0, 0, 0),
            BackgroundTransparency = 1,
            ClipsDescendants = true,
            LayoutOrder = 4,
        }, page)
        new("UIListLayout", {
            Padding = UDim.new(0, 8),
            SortOrder = Enum.SortOrder.LayoutOrder,
        }, flyContainer)

        local flyBtn = new("TextButton", {
            Size = UDim2.new(1, 0, 0, 34),
            BackgroundColor3 = Theme.RowBg,
            BorderSizePixel = 0,
            Text = "飞行 V3",
            Font = Enum.Font.GothamMedium,
            TextSize = 13,
            TextColor3 = Theme.Text,
            AutoButtonColor = false,
            LayoutOrder = 1,
        }, flyContainer)
        corner(flyBtn, 8)

        flyBtn.MouseEnter:Connect(function()
            TweenService:Create(flyBtn, TweenInfo.new(0.12), { BackgroundColor3 = Theme.Hover }):Play()
        end)
        flyBtn.MouseLeave:Connect(function()
            TweenService:Create(flyBtn, TweenInfo.new(0.12), { BackgroundColor3 = Theme.RowBg }):Play()
        end)

        flyBtn.MouseButton1Click:Connect(function()
            TweenService:Create(flyBtn, TweenInfo.new(0.08), { BackgroundColor3 = Theme.Accent }):Play()
            task.delay(0.12, function()
                TweenService:Create(flyBtn, TweenInfo.new(0.15), { BackgroundColor3 = Theme.RowBg }):Play()
            end)
            local ok, err = pcall(startFlyV3)
            if not ok then
                warn("[Fly V3] 运行失败: " .. tostring(err))
            end
        end)

        local flyExpanded = false
        flyHeaderBtn.MouseButton1Click:Connect(function()
            flyExpanded = not flyExpanded
            if flyExpanded then
                flyFolderIcon.Text = "📂"
                TweenService:Create(flyContainer, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, 42),
                }):Play()
            else
                flyFolderIcon.Text = "📁"
                TweenService:Create(flyContainer, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, 0),
                }):Play()
            end
        end)
    end

    -- 拖动 & 快捷键
    do
        local dragging, dragStart, startPos

        Main.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = Main.Position
                local conn
                conn = input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                        conn:Disconnect()
                    end
                end)
            end
        end)

        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                Main.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end)

        UserInputService.InputBegan:Connect(function(input, processed)
            if processed then return end
            if input.KeyCode == Enum.KeyCode.RightShift then
                if gui.Enabled and Main.Visible then
                    closeUI()
                else
                    gui.Enabled = true
                    Island.Visible = false
                    StatsPill.Visible = false
                    Main.Visible = true
                end
            end
        end)
    end

    selectTab(1)
end
