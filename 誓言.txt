}
local TARGET_PLACE_IDS  = {
    13822889,   -- Lumber Tycoon 2
    -- 最强战场的 PlaceId 请自行填入
}

local function isSupportedGame()
    for _, id in ipairs(TARGET_PLACE_IDS) do
        if game.PlaceId == id then return true, game.PlaceId end
    end
    local ok, info = pcall(function()
        return MarketplaceService:GetProductInfo(game.PlaceId)
    end)
    if ok and info and info.Name then
        for _, key in ipairs(TARGET_GAME_NAMES) do
            if string.find(info.Name, key, 1, true) then
                return true, info.Name
            end
        end
    end
    return false, nil
end

do
    local dragging, dragStart, startPos

    Main.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position

            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    conn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == Enum.KeyCode.RightShift then
            if gui.Enabled and Main.Visible then
                closeUI()
            else
                gui.Enabled = true
                Island.Visible = false
                StatsPill.Visible = false
                Main.Visible = true
            end
        end
    end)
end

local function loadMainScript()
    local supported = isSupportedGame()
    if not supported then
        return false, "当前游戏不在支持列表中，不加载主脚本"
    end
    local fn, compileErr = loadstring(MAIN_SCRIPT_SOURCE)
    if not fn then
        return false, "编译失败: " .. tostring(compileErr)
    end
    local ok, runErr = pcall(fn)
    if not ok then
        return false, "运行失败: " .. tostring(runErr)
    end
    return true
end

    local inst = Instance.new(class)

local function shake()
    local original = Card.Position
    local offsets = {10, -10, 8, -8, 6, -6, 4, -4, 2, -2, 0}
    for _, o in ipairs(offsets) do
        TweenService:Create(Card, TweenInfo.new(0.04), {
            Position = original + UDim2.fromOffset(o, 0),
        }):Play()
        task.wait(0.04)
    end
    Card.Position = original
end

local verifying = false
local function verify()
    if verifying then return end
    verifying = true

    local inputText = InputBox.Text
    local isEmpty   = (inputText == "")
    local isOwner   = (LocalPlayer.Name == OWNER_NAME)
    local isKeyOK   = (inputText == CORRECT_KEY)

    local passed = isKeyOK or (isEmpty and isOwner)

    if passed then
        TipLabel.TextColor3 = Theme.Accent
        TipLabel.Text = (isEmpty and isOwner) and "识别为开发者，正在检测游戏..." or "验证成功，正在检测游戏..."

        local ok, err = loadMainScript()

        if ok then
            TweenService:Create(Card, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
                Size = UDim2.fromOffset(280, 320),
                BackgroundTransparency = 1,
            }):Play()
            TweenService:Create(Title, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
            TweenService:Create(Subtitle, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
            TweenService:Create(InputHolder, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
            TweenService:Create(ConfirmBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
            TweenService:Create(BottomImage, TweenInfo.new(0.25), { ImageTransparency = 1 }):Play()
            TweenService:Create(Backdrop, TweenInfo.new(0.35), { BackgroundTransparency = 1 }):Play()
            task.wait(0.35)
            gui:Destroy()
        else
            TipLabel.TextColor3 = Theme.Red
            TipLabel.Text = tostring(err)
            warn("[Oath] " .. tostring(err))
            TweenService:Create(InputStroke, TweenInfo.new(0.15), { Color = Theme.Red }):Play()
            task.delay(0.6, function()
                TweenService:Create(InputStroke, TweenInfo.new(0.3), { Color = Theme.Stroke }):Play()
            end)
            verifying = false
        end
    else
        TipLabel.TextColor3 = Theme.Red
        TipLabel.Text = isEmpty and "请输入卡密" or "卡密错误，请重试"
        TweenService:Create(InputStroke, TweenInfo.new(0.15), { Color = Theme.Red }):Play()
        shake()
        task.delay(0.6, function()
            TweenService:Create(InputStroke, TweenInfo.new(0.3), { Color = Theme.Stroke }):Play()
        end)
        InputBox.Text = ""
        verifying = false
    end
end
