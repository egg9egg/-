local function uprightCFrame(hrp, cf)
    local pos = cf.Position
    local yRot = hrp and hrp.Orientation.Y or 0
    return CFrame.new(pos) * CFrame.Angles(0, math.rad(yRot), 0)
end
local function unlockPlayer()
    if lockPlayerConn then lockPlayerConn:Disconnect(); lockPlayerConn = nil end
end
local function lockPlayerAt(cf)
    unlockPlayer()

        if hrp then

            hrp.Velocity = Vector3.new(0,0,0)
            hrp.RotVelocity = Vector3.new(0,0,0)
        end
    end

end

-- ==================== 拖拽 ====================
local function getNextDragID() bai.dragIDCounter = bai.dragIDCounter + 1; return bai.dragIDCounter end
local DRAG_TOTAL = 60
local DRAG_SET_START = 28
local DRAG_SET_COUNT = 2
local DRAG_SET_INTERVAL = 0.01
local DRAG_REFRESH_INTERVAL = 0.01
local DRAG_AFTER_WAIT = 0.01

local function dragToPosition(model, targetCF, callback, waitFinish)
    if not model or not model.Parent then if callback then callback() end; return end
    local primary = model.PrimaryPart or model:FindFirstChild("WoodSection") or model:FindFirstChild("Main") or model:FindFirstChildWhichIsA("BasePart")
    if not primary then if callback then callback() end; return end
    if not model.PrimaryPart then model.PrimaryPart = primary end

    lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))
    local dragID = getNextDragID()
    local remote = rep.Interaction.ClientIsDragging

    pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
    pcall(function() remote:FireServer("Begin", model, dragID) end)

    local lastSet = DRAG_SET_START + DRAG_SET_COUNT - 1
    for i = 1, lastSet do
        if not model.Parent or not model.PrimaryPart then
            pcall(function() remote:FireServer("End", model, dragID) end)
            if callback then callback() end
            return
        end
        local shouldSet = (i >= DRAG_SET_START)
        pcall(function()
            remote:FireServer("Refresh", model, dragID)
            if shouldSet then
                rep.Interaction.ClientRequestOwnership:FireServer(primary)
                model:SetPrimaryPartCFrame(targetCF)
            end
        end)
        if i < lastSet then
            if shouldSet then task.wait(DRAG_SET_INTERVAL) else task.wait(DRAG_REFRESH_INTERVAL) end
        end
    end

    task.wait(DRAG_AFTER_WAIT)

    local dragEnded = false
    task.spawn(function()
        for i = lastSet + 1, DRAG_TOTAL do
            if not model.Parent or not model.PrimaryPart then break end
            pcall(function() remote:FireServer("Refresh", model, dragID) end)
            if i < DRAG_TOTAL then task.wait(DRAG_REFRESH_INTERVAL) end
        end
        pcall(function() remote:FireServer("End", model, dragID) end)
        dragEnded = true
    end)

    if waitFinish then
        local w = 0
        while w < 1.5 and not dragEnded do task.wait(0.05); w = w + 0.05 end
    end

    task.wait(0.4)
    local verify = 0
    while model.Parent and model.PrimaryPart and verify < 2 do
        local dist = (model.PrimaryPart.Position - targetCF.Position).Magnitude
        if dist <= 3 then break end
        verify = verify + 1
        lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))
        task.wait(0.15)
        local retryID = getNextDragID()
        pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
        pcall(function() remote:FireServer("Begin", model, retryID) end)
        for _ = 1, 10 do
            if not model.Parent then break end
            pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
            pcall(function() remote:FireServer("Refresh", model, retryID) end)
            pcall(function() model:SetPrimaryPartCFrame(targetCF) end)
            task.wait(0.05)
        end
        pcall(function() remote:FireServer("End", model, retryID) end)
        task.wait(0.15)
    end

    unlockPlayer()
    if callback then callback() end
end

local function sortTreeNames(list)
    local prio = {}
    for i, n in ipairs(TREE_PRIORITY) do prio[n] = i end
    table.sort(list, function(a, b)
        local pa, pb = prio[a], prio[b]
        if pa and pb then return pa < pb end
        if pa then return true end
        if pb then return false end
        return a < b
    end)
end

local function findUncutTree(treeClass)
    local playerPos = lp.Character and lp.Character.HumanoidRootPart.Position
    if not playerPos then return nil end
    local nearestDist, nearestWood, nearestTree = math.huge, nil, nil
    for _, region in pairs(ws:GetChildren()) do
        if region.Name == "TreeRegion" then
            for _, tree in pairs(region:GetChildren()) do
                if tree:IsA("Model") and tree:FindFirstChild("TreeClass") and tree.TreeClass.Value == treeClass then
                    local woodSections = {}
                    for _, child in pairs(tree:GetChildren()) do
                        if child.Name == "WoodSection" and child:IsA("BasePart") then table.insert(woodSections, child) end
                    end
                    if #woodSections >= 2 then
                        local dist = (woodSections[1].Position - playerPos).Magnitude
                        if dist < nearestDist then nearestDist, nearestWood, nearestTree = dist, woodSections[1], tree end
                    end
                end
            end
        end
    end
    return nearestWood, nearestTree
end

local function getTreeTopCFrame(treeModel)
    local topY, topCF = -math.huge, nil
    for _, child in pairs(treeModel:GetChildren()) do
        if child:IsA("BasePart") then
            local top = child.Position.Y + child.Size.Y / 2
            if top > topY then topY = top; topCF = CFrame.new(child.Position.X, top + 2, child.Position.Z) end
        end
    end
    if not topCF then topCF = treeModel:GetPivot() + Vector3.new(0, 2, 0) end
    return topCF
end

local function getTreeClassFromModel(treeModel)
    local tc = treeModel:FindFirstChild("TreeClass")
    if tc then return tc.Value end
    local name = treeModel.Name
    for _, suffix in ipairs({"SpookyNeon","Spooky","GoldSwampy","Cherry","CaveCrawler","MuckySewer"}) do
        if name:find(suffix) then return suffix end
    end
    return "Generic"
end

local axeConfigCache2 = {}
local function loadWeaponConfigPlan2(weaponName)
    if axeConfigCache2[weaponName] ~= nil then return axeConfigCache2[weaponName] end
    if weaponName == "OlReliable" or weaponName == "OIReliable" then
        local override = nil
        local wsTool = workspace:FindFirstChild(lp.Name)
        if wsTool then wsTool = wsTool:FindFirstChild("Tool") end
        if wsTool then
            local tn = wsTool:FindFirstChild("ToolName")
            if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                override = wsTool:FindFirstChild("AxeClassDamageOverride")
            end
        end
        if not override then
            local bp = lp:FindFirstChild("Backpack")
            if bp then
                for _, tool in pairs(bp:GetChildren()) do
                    if tool:IsA("Tool") then
                        local tn = tool:FindFirstChild("ToolName")
                        if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                            override = tool:FindFirstChild("AxeClassDamageOverride")
                            if override then break end
                        end
                    end
                end
            end
        end
        if override then
            local cfg = {Damage=nil, SwingCooldown=nil, Range=nil, SpecialTrees={}}
            for k, v in pairs(override:GetAttributes()) do
                if k == "Damage" then cfg.Damage = v
                elseif k == "SwingCooldown" then cfg.SwingCooldown = v
                elseif k == "Range" then cfg.Range = v
                else cfg.SpecialTrees[k] = {Damage = v} end
            end
            for _, c in pairs(override:GetChildren()) do
                if c:IsA("NumberValue") or c:IsA("IntValue") then
                    if c.Name == "Damage" then cfg.Damage = c.Value
                    elseif c.Name == "SwingCooldown" then cfg.SwingCooldown = c.Value
                    elseif c.Name == "Range" then cfg.Range = c.Value
                    else cfg.SpecialTrees[c.Name] = {Damage = c.Value} end
                end
            end
            if cfg.Damage or cfg.SwingCooldown or next(cfg.SpecialTrees) then
                axeConfigCache2[weaponName] = cfg
                return cfg
            end
        end
    end
    local function decompileSafe(s)
        if not s then return nil end
        if type(decompile) == "function" then
            local ok, src = pcall(decompile, s)
            if ok and type(src) == "string" and #src > 0 then return src end
        end
        if type(getscriptbytecode) == "function" then
            local ok, bc = pcall(getscriptbytecode, s)
            if ok and type(bc) == "string" then return bc end
        end
        return nil
    end
    local function parseNum(src, key)
        local v = src:match("[%w_%.]+%.?" .. key .. "%s*=%s*([%-]?[%d%.]+)")
        if not v then v = src:match(key .. "%s*=%s*([%-]?[%d%.]+)") end
        return tonumber(v)
    end
    local function parseSpecialTrees(src)
        local out = {}
        for tn, body in src:gmatch("SpecialTrees%.([%w_]+)%s*=%s*{(.-)}") do
            local cfg = {}
            cfg.Damage = tonumber(body:match("Damage%s*=%s*([%-]?[%d%.]+)"))
            cfg.SwingCooldown = tonumber(body:match("SwingCooldown%s*=%s*([%-]?[%d%.]+)"))
            if cfg.Damage then out[tn] = cfg end
        end
        return out
    end
    local function findAxeModule(name)
        local want = name.."AxeClass"
        local la = rep:FindFirstChild("LoadedAssets")
        if la then local m = la:FindFirstChild(want); if m then return m end end
        if type(getloadedmodules) == "function" then
            for _, m in ipairs(getloadedmodules()) do
                if m:IsA("ModuleScript") and (m.Name == want or m:GetFullName():sub(-#want) == want) then return m end
            end
        end
        return nil
    end
    local module = findAxeModule(weaponName)
    if not module then axeConfigCache2[weaponName] = nil; return nil end
    local src = decompileSafe(module)
    if not src then axeConfigCache2[weaponName] = nil; return nil end
    local cfg = {
        Damage = parseNum(src, "Damage"),
        SwingCooldown = parseNum(src, "SwingCooldown"),
        Range = parseNum(src, "Range"),
        SpecialTrees = parseSpecialTrees(src),
    }
    if not cfg.Damage or not cfg.SwingCooldown then axeConfigCache2[weaponName] = nil; return nil end
    cfg.Range = cfg.Range or 10
    axeConfigCache2[weaponName] = cfg
    return cfg
end

local function bringTreePlan2(treeClass)
    local weaponName, weaponConfig, damage = selectBestWeaponPlan2(treeClass)
    if not weaponName then notify("誓言", "自动选武器失败", 3); return false end
    local req = {Flame=200000, BlueFlame=200000, Celestial=200000, Void=200000, Ice=200000, Magma=200000, RainbowFlame=200000, Radioactive=200000, Shine=200000, LoneCave=200000, Spirit=200000}
    if req[treeClass] and damage < req[treeClass] then notify("誓言", string.format("伤害不足！需要至少 %.0f", req[treeClass]), 4); return false end
    local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown
    local woodSec, tree = findUncutTree(treeClass)
    if not woodSec then notify("誓言", "附近没有可砍的树", 3); return false end
    local minX, maxX = -543, -27.8
    local minY, maxY = -196.7, -148.2
    local minZ, maxZ = -2961.7, -2668.7
    local tp_ = woodSec.Position
    if tp_.X >= minX and tp_.X <= maxX and tp_.Y >= minY and tp_.Y <= maxY and tp_.Z >= minZ and tp_.Z <= maxZ then
        notify("誓言", "树在危险区域内，已删除", 5)
        if tree and tree.Parent then tree:Destroy() end
        return false
    end
    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local tf = lp:FindFirstChild("ToolFolder")
        if tf then local wo = tf:FindFirstChild(weaponName); tool = wo and (wo:IsA("Tool") and wo or wo.Value) or nil end
    end
    if not tool then notify("誓言", "未找到工具: "..weaponName, 3); return false end
    local cutEvent = tree:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无法获取CutEvent", 3); return false end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local playerOriginalCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
    if not playerOriginalCF then notify("誓言", "无法获取玩家位置", 3); return false end
    local tempPart = Instance.new("Part")
    tempPart.Name = "TempMarker"; tempPart.Anchored = true; tempPart.CanCollide = true
    tempPart.Shape = Enum.PartType.Block; tempPart.Size = Vector3.new(0.001,0.0001,0.0001)
    tempPart.Color = Color3.fromRGB(255,0,0); tempPart.Material = Enum.Material.Neon
    tempPart.CFrame = CFrame.new(woodSec.Position + Vector3.new(8, 2.5, 0)); tempPart.Parent = workspace
    task.wait(0.02)
    lockPlayerAt(getTreeTopCFrame(tree))
    local fallenLog, existing = nil, {}
    for _, log in pairs(ws.LogModels:GetChildren()) do existing[log] = true end
    local treeCenter = woodSec.Position
    local stopCutting, timeoutReached = false, false
    local sendTask = task.spawn(function()
        while not stopCutting do
            pcall(remote.FireServer, remote, cutEvent, {height=0.3, faceVector=Vector3.new(-1,0,0), cooldown=cooldown, sectionId=1, hitPoints=damage, tool=tool, cuttingClass="Axe"})
            task.wait(0.0001)
        end
    end)
    local timeoutTask = task.spawn(function()
        task.wait(15)
        if not stopCutting then timeoutReached = true; stopCutting = true end
    end)
    while not fallenLog and not stopCutting do
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not existing[log] and log:FindFirstChild("Owner") and log.Owner.Value==lp and log:FindFirstChild("TreeClass") and log.TreeClass.Value==treeClass then
                if (log:GetPivot().Position - treeCenter).Magnitude < 100 then fallenLog = log; break end
            end
        end
        task.wait(0.01)
    end
    stopCutting = true; task.cancel(sendTask); task.cancel(timeoutTask)
    if timeoutReached or not fallenLog then
        notify("誓言", "砍树超时或木头未掉落", 3)
        if tempPart and tempPart.Parent then tempPart:Destroy() end
        safeTeleport(playerOriginalCF)
        return false
    end
    local primary = fallenLog.PrimaryPart or fallenLog:FindFirstChild("WoodSection") or fallenLog:FindFirstChild("Main") or fallenLog:FindFirstChildWhichIsA("BasePart")
    if not primary then
        notify("誓言", "无法定位木头", 3)
        if tempPart and tempPart.Parent then tempPart:Destroy() end
        safeTeleport(playerOriginalCF)
        return false
    end
    local logPos = primary.Position
    if logPos.X >= minX and logPos.X <= maxX and logPos.Y >= minY and logPos.Y <= maxY and logPos.Z >= minZ and logPos.Z <= maxZ then
        notify("誓言", "木头掉落在危险区域，已删除", 5)
        if fallenLog and fallenLog.Parent then fallenLog:Destroy() end
        if tempPart and tempPart.Parent then tempPart:Destroy() end
        safeTeleport(playerOriginalCF)
        return false
    end
    if not fallenLog.PrimaryPart then fallenLog.PrimaryPart = primary end
    lockPlayerAt(primary.CFrame + Vector3.new(0,3,0))
    task.wait(0.15)
    for _ = 1, 3 do pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary); task.wait(0.01) end
    local done = false
    dragToPosition(fallenLog, playerOriginalCF, function() done = true end, 10)
    local t = 0
    while not done and t < 6 do task.wait(0.05); t = t + 0.05 end
    if tempPart and tempPart.Parent then tempPart:Destroy() end
    safeTeleport(playerOriginalCF)
    notify("誓言", "树已带来", 2)
    return true
end

local isSelectingSawmill = false
local sawmillSelectConn = nil
local function selectSawmill()
    if isSelectingSawmill then notify("誓言", "正在等待选择中...", 2); return end
    isSelectingSawmill = true
    notify("誓言", "请点击锯木机任意部件", 3)
    if sawmillSelectConn then sawmillSelectConn:Disconnect() end
    sawmillSelectConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then notify("誓言", "未选中任何物体", 2); return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then notify("誓言", "请点击一个模型", 2); return end
        local function isSawmill(root)
            local attr = root:GetAttribute("ItemName")
            if attr and type(attr) == "string" and attr:lower():find("saw") then return true end
            for _, child in pairs(root:GetDescendants()) do
                if child.Name == "ItemName" then local val = child.Value; if val and type(val) == "string" and val:lower():find("saw") then return true end end
            end
            if root.Name and root.Name:lower():find("saw") then return true end
            local typeVal = root:FindFirstChild("Type")
            if typeVal and typeVal.Value == "Sawmill" then return true end
            return false
        end
        if not isSawmill(model) then notify("誓言", "这不是锯木机", 3); return end
        local function findPart(root)
            for _, child in pairs(root:GetDescendants()) do if child.Name == "Conveyor" and child:IsA("BasePart") then return child, "Conveyor" end end
            for _, child in pairs(root:GetDescendants()) do if child.Name == "Main" and child:IsA("BasePart") then return child, "Main" end end
            return nil, nil
        end
        local part, partType = findPart(model)
        if not part then notify("誓言", "找不到 Conveyor 或 Main 部件", 3); return end
        bai.cachedSawmill = model
        bai.cachedSawmillPos = part.CFrame
        if sawmillSelectConn then sawmillSelectConn:Disconnect(); sawmillSelectConn = nil end
        isSelectingSawmill = false
        notify("誓言", "✅ 已记忆锯木机位置（" .. partType .. "）", 3)
    end)
end

local isDecomposeActive = false
local decomposeConn = nil
local function startDecomposeSelection()
    if isDecomposeActive then notify("誓言", "分解选择已激活", 2); return end
    if not bai.cachedSawmill or not bai.cachedSawmill.Parent then notify("誓言", "请先使用 '选择锯木机' 按钮设置一个锯木机", 3); return end
    if not bai.cachedSawmillPos then notify("誓言", "锯木机位置无效", 3); return end
    isDecomposeActive = true
    notify("誓言", "请点击一棵树", 3)
    decomposeConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then notify("誓言", "未选中任何物体", 2); return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then notify("誓言", "请点击树干", 2); return end
        local hasWood = false
        for _, child in pairs(model:GetChildren()) do if child.Name == "WoodSection" then hasWood = true end end
        if not (hasWood and model:FindFirstChild("CutEvent")) then notify("誓言", "这不是一棵树", 2); return end
        if decomposeConn then decomposeConn:Disconnect(); decomposeConn = nil end
        isDecomposeActive = false
        notify("誓言", "已选中树，开始分解...", 3)
        task.spawn(function()
            local treeClass = getTreeClassFromModel(model)
            local weaponName, weaponConfig, damage
            if selectedPlan == 2 then
                weaponName, weaponConfig, damage = selectBestWeaponPlan2(treeClass)
            else
                weaponName, weaponConfig, damage = selectBestWeaponPlan1(treeClass)
            end
            if not weaponName then notify("誓言", "自动选武器失败", 3); return end
            local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown
            local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
            if not tool then
                local tf = lp:FindFirstChild("ToolFolder")
                if tf then local wo = tf:FindFirstChild(weaponName); tool = wo and (wo:IsA("Tool") and wo or wo.Value) or nil end
            end
            if not tool then notify("誓言", "未找到工具: "..weaponName, 3); return end
            local cutEvent = model:FindFirstChild("CutEvent")
            if not cutEvent then notify("誓言", "无法获取CutEvent", 3); return end
            local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")

            local sections = {}
            local rootSection = nil
            for _, child in pairs(model:GetChildren()) do
                if child.Name == "WoodSection" and child:IsA("BasePart") and child:FindFirstChild("ID") then
                    local id = child.ID.Value
                    if id >= 2 then table.insert(sections, {id = id, part = child})
                    elseif id == 1 then rootSection = child end
                end
            end
            if #sections == 0 then notify("誓言", "该树没有可分解的部分（ID≥2）", 3); return end
            table.sort(sections, function(a, b) return a.id > b.id end)

            local initialLogs = {}
            for _, log in pairs(ws.LogModels:GetChildren()) do initialLogs[log] = true end
            local tasks = {}
            notify("誓言", string.format("开始分解 %d 个部分", #sections), 2)

            for _, sec in ipairs(sections) do
                local part = sec.part
                local sectionId = sec.id
                local stopFlag = false
                local taskId = task.spawn(function()
                    while not stopFlag do
                        pcall(remote.FireServer, remote, cutEvent, {height=0.11, faceVector=Vector3.new(-1,0,0), cooldown=cooldown, sectionId=sectionId, hitPoints=damage, tool=tool, cuttingClass="Axe"})
                        task.wait(0.01)
                        if not part or not part.Parent then break end
                    end
                end)
                tasks[taskId] = true
            end

            local function hasHighIDSections()
                for _, child in pairs(model:GetChildren()) do
                    if child.Name == "WoodSection" and child:FindFirstChild("ID") and child.ID.Value >= 2 then return true end
                end
                return false
            end

            local startTime = tick()
            while hasHighIDSections() and tick() - startTime < 120 do task.wait(0.01) end
            for taskId in pairs(tasks) do task.cancel(taskId) end
            if hasHighIDSections() then notify("誓言", "停止", 3); return end

            local allLogs = {}
            for _, log in pairs(ws.LogModels:GetChildren()) do
                if not initialLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp then table.insert(allLogs, log) end
            end
            if #allLogs == 0 then notify("誓言", "未找到木头（可能掉落失败），停止", 3); return end

            local targetPos = bai.cachedSawmillPos + Vector3.new(0.7, 0, 0)
            for idx, log in ipairs(allLogs) do
                if log and log.Parent then
                    local primary = log.PrimaryPart or log:FindFirstChild("WoodSection") or log:FindFirstChild("Main") or log:FindFirstChildWhichIsA("BasePart")
                    if primary then
                        if not log.PrimaryPart then log.PrimaryPart = primary end
                        safeTeleportNoWait(primary.CFrame + Vector3.new(0, 2, 0))
                        for _ = 1, 3 do pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end); task.wait(0.01) end
                        for i = 1, 30 do
                            pcall(function() rep.Interaction.ClientIsDragging:FireServer(log, DRAG_SECRET); log:SetPrimaryPartCFrame(targetPos) end)
                            task.wait(0.01)
                        end
                    end
                end
            end

            if rootSection and rootSection.Parent then
                if not model.PrimaryPart then model.PrimaryPart = rootSection end
                safeTeleport(rootSection.CFrame + Vector3.new(0, 2, 0))
                dragToPosition(model, targetPos)
            end
            notify("誓言", "分解完成", 3)
        end)
    end)
end

local function createTreeDropdown(parent, options, default, order, onSelect)
    local HEADER_H, SEARCH_H, ITEM_H, MAX_LIST_H = 32, 26, 24, 180
    local container = new("Frame",{Size=UDim2.new(1, 0, 0, HEADER_H), BackgroundTransparency=1, LayoutOrder=order, ClipsDescendants=false, ZIndex=1}, parent)
    local header = new("Frame",{Size=UDim2.new(1, 0, 0, HEADER_H), BackgroundColor3=T.RowBg, BorderSizePixel=0, ZIndex=2}, container)
    crn(header, 6); strk(header, Color3.fromRGB(48,48,52), 1, 0.3)
    local cur = default or options[1]
    new("TextLabel",{Position=UDim2.new(0, 12, 0, 0), Size=UDim2.new(0.35, 0, 0, HEADER_H), BackgroundTransparency=1, Text="选择树木种类", Font=Enum.Font.GothamMedium, TextSize=12, TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left, ZIndex=3}, header)
    local valueLbl = new("TextLabel",{Position=UDim2.new(0.4, 0, 0, 0), Size=UDim2.new(0.5, -12, 0, HEADER_H), BackgroundTransparency=1, Text=tostring(cur), Font=Enum.Font.Gotham, TextSize=11, TextColor3=T.Accent, TextXAlignment=Enum.TextXAlignment.Right, TextTruncate=Enum.TextTruncate.AtEnd, ZIndex=3}, header)
    local plusBtn = new("TextButton",{Size=UDim2.fromOffset(22, 22), Position=UDim2.new(1, -28, 0.5, -11), BackgroundColor3=Color3.fromRGB(48, 48, 52), BorderSizePixel=0, Text="+", Font=Enum.Font.GothamBold, TextSize=16, TextColor3=T.Text, AutoButtonColor=false, ZIndex=3}, header)
    crn(plusBtn, 4)
    local listFrame = new("Frame",{Position=UDim2.new(0, 0, 0, HEADER_H), Size=UDim2.new(1, 0, 0, 0), BackgroundColor3=T.Panel, BorderSizePixel=0, ClipsDescendants=true, Visible=false, ZIndex=10}, container)
    crn(listFrame, 6)
    local searchBox = new("TextBox",{Position=UDim2.new(0, 4, 0, 4), Size=UDim2.new(1, -8, 0, SEARCH_H - 4), BackgroundColor3=Color3.fromRGB(24, 25, 29), BorderSizePixel=0, Text="", PlaceholderText="搜索...", PlaceholderColor3=T.Sub, Font=Enum.Font.Gotham, TextSize=11, TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left, ClearTextOnFocus=false, ZIndex=11}, listFrame)
    crn(searchBox, 4)
    local scrollFrame = new("ScrollingFrame",{Position=UDim2.new(0, 4, 0, SEARCH_H + 4), Size=UDim2.new(1, -8, 0, 0), BackgroundTransparency=1, BorderSizePixel=0, ScrollBarThickness=3, ScrollBarImageColor3=T.Stroke, CanvasSize=UDim2.new(0, 0, 0, 0), AutomaticCanvasSize=Enum.AutomaticSize.Y, ZIndex=11}, listFrame)
    new("UIListLayout",{ Padding=UDim.new(0, 2), SortOrder=Enum.SortOrder.LayoutOrder }, scrollFrame)

    local expanded = false
    local function buildItems(filterText)
        for _, ch in ipairs(scrollFrame:GetChildren()) do if ch:IsA("TextButton") then ch:Destroy() end end
        filterText = (filterText or ""):lower()
        local count = 0
        for i, opt in ipairs(options) do
            local txt = tostring(opt)
            if filterText == "" or txt:lower():find(filterText, 1, true) then
                count = count + 1
                local b = new("TextButton",{Size=UDim2.new(1, 0, 0, ITEM_H - 2), BackgroundTransparency=1, Text="   "..txt, Font=Enum.Font.Gotham, TextSize=11, TextColor3=T.Text, TextXAlignment=Enum.TextXAlignment.Left, LayoutOrder=i, AutoButtonColor=false, ZIndex=12}, scrollFrame)
                b.MouseButton1Click:Connect(function()
                    cur = opt; valueLbl.Text = tostring(cur)
                    if onSelect then pcall(onSelect, cur) end
                    expand(false)
                end)
            end
        end
        return count
    end
    function expand(state)
        expanded = state
        if state then
            container.ZIndex = 100
            plusBtn.Text = "−"; listFrame.Visible = true
            local cnt = buildItems(searchBox.Text)
            local listH = SEARCH_H + 8 + math.min(cnt * ITEM_H, MAX_LIST_H)
            TS:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, listH) }):Play()
            TS:Create(scrollFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, -8, 0, math.min(cnt * ITEM_H, MAX_LIST_H)) }):Play()
            TS:Create(container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, HEADER_H + listH) }):Play()
        else
            container.ZIndex = 1; plusBtn.Text = "+"
            TS:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 0) }):Play()
            TS:Create(container, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, HEADER_H) }):Play()
            task.delay(0.2, function() listFrame.Visible = false end)
        end
    end
    plusBtn.MouseButton1Click:Connect(function() expand(not expanded) end)
    searchBox:GetPropertyChangedSignal("Text"):Connect(function()
        if expanded then
            local cnt = buildItems(searchBox.Text)
            local listH = SEARCH_H + 8 + math.min(cnt * ITEM_H, MAX_LIST_H)
            TS:Create(listFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, listH) }):Play()
            TS:Create(scrollFrame, TweenInfo.new(0.15), { Size = UDim2.new(1, -8, 0, math.min(cnt * ITEM_H, MAX_LIST_H)) }):Play()
            TS:Create(container, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, HEADER_H + listH) }):Play()
        end
    end)
    return container, function() if expanded then expand(false) end end
end
