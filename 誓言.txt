local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in next, props do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end
local function corner(parent, r)
    return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, parent)
end
local function stroke(parent, color, thickness, transparency)
    return new("UIStroke", {
        Color = color or Theme.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function makeActionButton(parent, text, order, callback)
    local btn = new("TextButton", {
        Size = UDim2.new(1, 0, 0, 34),
        BackgroundColor3 = Theme.RowBg,
        BorderSizePixel = 0,
        Text = text,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        AutoButtonColor = false,
        LayoutOrder = order,
    }, parent)
    corner(btn, 8)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = Theme.Hover }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = Theme.RowBg }):Play()
    end)
    btn.MouseButton1Click:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.08), { BackgroundColor3 = Theme.Accent }):Play()
        task.delay(0.12, function()
            TweenService:Create(btn, TweenInfo.new(0.15), { BackgroundColor3 = Theme.RowBg }):Play()
        end)
        if callback then
            local ok, err = pcall(callback)
            if not ok then warn("[按钮错误] " .. tostring(err)) end
        end
    end)
    return btn
end

local function makeSliderCard(parent, text, minValue, maxValue, default, order, callback)
    local row = new("Frame", {
        Size = UDim2.new(1, 0, 0, 52),
        BackgroundColor3 = Theme.RowBg,
        BorderSizePixel = 0,
        LayoutOrder = order,
    }, parent)
    corner(row, 8)
    new("TextLabel", {
        Position = UDim2.new(0, 12, 0, 8),
        Size = UDim2.new(0.55, 0, 0, 16),
        BackgroundTransparency = 1,
        Text = text,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)
    local valueBox = new("TextBox", {
        Position = UDim2.new(0.55, 0, 0, 6),
        Size = UDim2.new(0.45, -12, 0, 20),
        BackgroundColor3 = Color3.fromRGB(24, 25, 29),
        BorderSizePixel = 0,
        Text = tostring(default),
        Font = Enum.Font.GothamBold,
        TextSize = 12,
        TextColor3 = Theme.Accent,
        TextXAlignment = Enum.TextXAlignment.Center,
        ClearTextOnFocus = false,
    }, row)
    corner(valueBox, 6)
    local track = new("Frame", {
        Position = UDim2.new(0, 12, 0, 36),
        Size = UDim2.new(1, -24, 0, 4),
        BackgroundColor3 = Color3.fromRGB(55, 57, 64),
        BorderSizePixel = 0,
    }, row)
    corner(track, 2)
    local span = math.max(maxValue - minValue, 1)
    local ratio = math.clamp((default - minValue) / span, 0, 1)
    local fill = new("Frame", {
        Size = UDim2.new(ratio, 0, 1, 0),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
    }, track)
    corner(fill, 2)
    local knob = new("Frame", {
        Size = UDim2.fromOffset(12, 12),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(ratio, 0, 0.5, 0),
        BackgroundColor3 = Color3.fromRGB(240, 240, 245),
        BorderSizePixel = 0,
        ZIndex = 2,
    }, track)
    corner(knob, 6)
    local hit = new("TextButton", {
        Size = UDim2.new(1, 16, 3, 16),
        Position = UDim2.new(0, -8, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        BackgroundTransparency = 1,
        Text = "",
        ZIndex = 3,
    }, track)
    local dragging, focused = false, false
    local currentValue = default
    local function setValue(value, updateBox)
        value = math.clamp(math.floor(value + 0.5), minValue, maxValue)
        currentValue = value
        local rel = (value - minValue) / span
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, 0, 0.5, 0)
        if updateBox and not focused then valueBox.Text = tostring(value) end
        if callback then pcall(callback, value) end
    end
    setValue(default, false)
    local function applyFromX(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        setValue(minValue + rel * span, true)
    end
    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; applyFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            applyFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    valueBox.Focused:Connect(function() focused = true end)
    valueBox.FocusLost:Connect(function()
        focused = false
        local num = tonumber(valueBox.Text)
        if not num then valueBox.Text = tostring(currentValue); return end
        setValue(num, true)
    end)
end

local function makeDropdown(parent, text, options, default, order, callback)
    local container = new("Frame", {
        Size = UDim2.new(1, 0, 0, 34),
        BackgroundColor3 = Theme.RowBg,
        BorderSizePixel = 0,
        LayoutOrder = order,
    }, parent)
    corner(container, 8)
    local current = default or options[1]
    local display = new("TextButton", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = text .. ": " .. tostring(current),
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        TextXAlignment = Enum.TextXAlignment.Left,
        AutoButtonColor = false,
    }, container)
    new("UIPadding", { PaddingLeft = UDim.new(0, 12) }, display)
    local listFrame = new("Frame", {
        Size = UDim2.new(1, 0, 0, 0),
        Position = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Theme.Panel,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Visible = false,
        ZIndex = 10,
    }, container)
    corner(listFrame, 8)
    new("UIListLayout", { Padding = UDim.new(0, 2) }, listFrame)
    local expanded = false
    local function buildList()
        for _, child in ipairs(listFrame:GetChildren()) do
            if child:IsA("TextButton") then child:Destroy() end
        end
        for i, opt in ipairs(options) do
            local btn = new("TextButton", {
                Size = UDim2.new(1, 0, 0, 26),
                BackgroundTransparency = 1,
                Text = tostring(opt),
                Font = Enum.Font.Gotham,
                TextSize = 12,
                TextColor3 = Theme.Text,
                TextXAlignment = Enum.TextXAlignment.Left,
                LayoutOrder = i,
                AutoButtonColor = false,
            }, listFrame)
            new("UIPadding", { PaddingLeft = UDim.new(0, 12) }, btn)
            btn.MouseEnter:Connect(function() btn.BackgroundTransparency = 0.1; btn.BackgroundColor3 = Theme.Hover end)
            btn.MouseLeave:Connect(function() btn.BackgroundTransparency = 1 end)
            btn.MouseButton1Click:Connect(function()
                current = opt
                display.Text = text .. ": " .. tostring(current)
                expanded = false
                TweenService:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 0) }):Play()
                task.delay(0.2, function() listFrame.Visible = false end)
                if callback then pcall(callback, current) end
            end)
        end
        listFrame.Size = UDim2.new(1, 0, 0, math.min(#options * 28, 200))
    end
    display.MouseButton1Click:Connect(function()
        expanded = not expanded
        if expanded then
            buildList()
            listFrame.Visible = true
            listFrame.Size = UDim2.new(1, 0, 0, 0)
            TweenService:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, math.min(#options * 28, 200)) }):Play()
        else
            TweenService:Create(listFrame, TweenInfo.new(0.2), { Size = UDim2.new(1, 0, 0, 0) }):Play()
            task.delay(0.2, function() listFrame.Visible = false end)
        end
    end)
end

local SectionObject = {}
SectionObject.__index = SectionObject
function SectionObject.new(container, title)
    local self = setmetatable({}, SectionObject)
    self.orderCounter = 0
    self.frame = new("Frame", {
        Size = UDim2.new(1, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.Y,
        BackgroundColor3 = Theme.Panel,
        BorderSizePixel = 0,
    }, container)
    corner(self.frame, 8)
    new("UIPadding", {
        PaddingTop = UDim.new(0, 8),
        PaddingBottom = UDim.new(0, 8),
        PaddingLeft = UDim.new(0, 8),
        PaddingRight = UDim.new(0, 8),
    }, self.frame)
    new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, self.frame)
    if title and title ~= "" then self:Label(title) end
    return self
end
function SectionObject:Label(text)
    self.orderCounter = self.orderCounter + 1
    return makeLabel(self.frame, text, self.orderCounter)
end
function SectionObject:Button(text, callback)
    self.orderCounter = self.orderCounter + 1
    return makeActionButton(self.frame, text, self.orderCounter, callback)
end
function SectionObject:Toggle(text, id, default, callback)
    self.orderCounter = self.orderCounter + 1
    return makeToggleButton(self.frame, text, self.orderCounter, default, callback)
end
function SectionObject:Slider(text, id, default, min, max, rounded, callback)
    self.orderCounter = self.orderCounter + 1
    return makeSliderCard(self.frame, text, min, max, default, self.orderCounter, callback)
end
function SectionObject:Dropdown(text, id, options, callback)
    self.orderCounter = self.orderCounter + 1
    return makeDropdown(self.frame, text, options, options[1], self.orderCounter, callback)
end
function SectionObject:Textbox(text, id, placeholder, callback)
    self.orderCounter = self.orderCounter + 1
    return makeTextbox(self.frame, text, placeholder, self.orderCounter, callback)
end

        icon.Text = expanded and "📂" or "📁"
        TweenService:Create(content, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(1, 0, 0, expanded and content.AbsoluteSize.Y or 0),
        }):Play()
    end)
    content.AutomaticSize = Enum.AutomaticSize.Y
    content.Size = UDim2.new(1, 0, 0, 0)

            TipLabel.Text = "验证成功，正在加载..."
            local ok, err = pcall(function()
                if _G.buildMainPanel then _G.buildMainPanel() end
            end)
            if ok then
                TweenService:Create(Card, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
                    Size = UDim2.fromOffset(280, 320), BackgroundTransparency = 1,
                }):Play()
                TweenService:Create(Title, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
                TweenService:Create(Subtitle, TweenInfo.new(0.25), { TextTransparency = 1 }):Play()
                TweenService:Create(InputHolder, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
                TweenService:Create(ConfirmBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
                TweenService:Create(Backdrop, TweenInfo.new(0.35), { BackgroundTransparency = 1 }):Play()
                task.wait(0.35)
                keyGui:Destroy()
            else
                TipLabel.TextColor3 = KeyTheme.Red
                TipLabel.Text = "主脚本加载失败"
                warn("[Oath] " .. tostring(err))
                verifying = false
            end
        else
            TipLabel.TextColor3 = KeyTheme.Red
            TipLabel.Text = "卡密错误，请重试"
            InputBox.Text = ""
            verifying = false
        end
    end

    ConfirmBtn.MouseButton1Click:Connect(verify)
    InputBox.FocusLost:Connect(function(enterPressed) if enterPressed then verify() end end)
    task.wait(0.3)
    pcall(function() InputBox:CaptureFocus() end)
end

_G.buildMainPanel = function()
    print("[" .. GAME_TITLE .. "] " .. VERSION .. " 已加载")

    local gui = new("ScreenGui", {
        Name = "OathPanel",
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset = true,
        DisplayOrder = 100,
    }, PlayerGui)

    local Main = new("Frame", {
        Size = UDim2.fromOffset(620, 400),
        Position = UDim2.fromScale(0.5, 0.5),
        AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = Theme.Bg,
        BorderSizePixel = 0,
        ClipsDescendants = true,
    }, gui)
    corner(Main, 16)
    stroke(Main, Theme.Stroke, 1)

    local Sidebar = new("Frame", {
        Size = UDim2.new(0, 150, 1, 0),
        BackgroundColor3 = Theme.Sidebar,
        BorderSizePixel = 0,
    }, Main)
    corner(Sidebar, 16)

    new("TextLabel", {
        Size = UDim2.new(0, 130, 0, 20),
        Position = UDim2.new(0, 14, 0, 6),
        BackgroundTransparency = 1,
        Text = GAME_TITLE,
        Font = Enum.Font.GothamBold,
        TextSize = 16,
        TextColor3 = Theme.Accent,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, Sidebar)

    local tabs = { "信息", "功能", "脚本", "设置", "关于" }
    local tabButtons = {}
    local pageContainer = new("Frame", {
        Position = UDim2.new(0, 150, 0, 0),
        Size = UDim2.new(1, -150, 1, 0),
        BackgroundTransparency = 1,
    }, Main)

    local pages = {}
    for i, name in ipairs(tabs) do
        local page = new("ScrollingFrame", {
            Name = name,
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            BorderSizePixel = 0,
            ScrollBarThickness = 3,
            ScrollBarImageColor3 = Theme.Stroke,
            CanvasSize = UDim2.new(0, 0, 0, 0),
            AutomaticCanvasSize = Enum.AutomaticSize.Y,
            Visible = false,
        }, pageContainer)
        new("UIListLayout", { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder }, page)
        new("UIPadding", {
            PaddingTop = UDim.new(0, 48),
            PaddingBottom = UDim.new(0, 10),
            PaddingLeft = UDim.new(0, 14),
            PaddingRight = UDim.new(0, 14),
        }, page)
        pages[i] = page
    end

    local holder = new("Frame", {
        Position = UDim2.new(0, 0, 0, 56),
        Size = UDim2.new(1, 0, 1, -64),
        BackgroundTransparency = 1,
    }, Sidebar)
    new("UIListLayout", { Padding = UDim.new(0, 4), HorizontalAlignment = Enum.HorizontalAlignment.Center }, holder)
    for i, name in ipairs(tabs) do
        local btn = new("TextButton", {
            Size = UDim2.new(1, 0, 0, 32),
            BackgroundTransparency = 1,
            Text = "",
            AutoButtonColor = false,
            LayoutOrder = i,
        }, holder)
        corner(btn, 16)
        local lbl = new("TextLabel", {
            Position = UDim2.new(0, 14, 0, 0),
            Size = UDim2.new(1, -14, 1, 0),
            BackgroundTransparency = 1,
            Text = name,
            Font = Enum.Font.GothamMedium,
            TextSize = 13,
            TextColor3 = Theme.Sub,
            TextXAlignment = Enum.TextXAlignment.Left,
        }, btn)
        tabButtons[i] = { button = btn, label = lbl }
        btn.MouseButton1Click:Connect(function()
            for j, p in ipairs(pages) do
                p.Visible = (j == i)
                tabButtons[j].label.TextColor3 = (j == i) and Theme.Text or Theme.Sub
            end
        end)
    end
    pages[1].Visible = true
    tabButtons[1].label.TextColor3 = Theme.Text

    local info = pages[1]
    local infoCard = new("Frame", {
        Size = UDim2.new(1, 0, 0, 100),
        BackgroundColor3 = Theme.Panel,
        BorderSizePixel = 0,
    }, info)
    corner(infoCard, 10)
    new("TextLabel", {
        Size = UDim2.new(1, -20, 0, 30),
        Position = UDim2.new(0, 10, 0, 10),
        BackgroundTransparency = 1,
        Text = "木材大亨2 · 誓言面板",
        Font = Enum.Font.GothamBold,
        TextSize = 14,
        TextColor3 = Theme.Text,
    }, infoCard)
    new("TextLabel", {
        Size = UDim2.new(1, -20, 0, 20),
        Position = UDim2.new(0, 10, 0, 40),
        BackgroundTransparency = 1,
        Text = "作者：是我太傻   QQ：3767465890   群：1109682737",
        Font = Enum.Font.Gotham,
        TextSize = 11,
        TextColor3 = Theme.Sub,
    }, infoCard)

    -- ★ 把 hiddenGui 里的 UI 转移到功能分区
    for _, child in ipairs(functionContainer:GetChildren()) do
        if child:IsA("GuiObject") then
            child.Parent = pages[2]
        end
    end
    hiddenGui:Destroy()

    for i = 3, 5 do
        new("TextLabel", {
            Size = UDim2.new(1, 0, 0, 30),
            BackgroundTransparency = 1,
            Text = tabs[i] .. " 分区",
            Font = Enum.Font.GothamBold,
            TextSize = 14,
            TextColor3 = Theme.Sub,
        }, pages[i])
    end

    UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == Enum.KeyCode.RightShift then
            gui.Enabled = not gui.Enabled
        end
    end)

    local dragging, dragStart, startPos
    Main.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    print("[Oath] 主面板创建完成")
end
