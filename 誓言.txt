local function new(class, props, parent)
    local inst = Instance.new(class)
    for k, v in next, props do inst[k] = v end
    if parent then inst.Parent = parent end
    return inst
end

local function corner(parent, r)
    return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, parent)
end

local function stroke(parent, color, thickness, transparency)
    return new("UIStroke", {
        Color = color or Theme.Stroke,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, parent)
end

local function makeWinDot(color, x, tip, callback)
    local btn = new("TextButton", {
        Size = UDim2.fromOffset(12, 12),
        Position = UDim2.new(0, x, 0, 1),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
        Text = "",
        AutoButtonColor = false,
        ZIndex = 10,
    }, BtnHolder)
    corner(btn, 6)
    btn.MouseEnter:Connect(function()
        btn.BackgroundColor3 = color:Lerp(Color3.new(1, 1, 1), 0.25)
    end)
    btn.MouseLeave:Connect(function()
        btn.BackgroundColor3 = color
    end)
    btn.MouseButton1Click:Connect(callback)
    btn.Name = tip
    return btn
end

local function makeSliderCard(parent, label, maxValue, default, order, onChange)
    local row = new("Frame", {
        Size = UDim2.new(1, 0, 0, 52),
        BackgroundColor3 = Theme.Panel,
        BorderSizePixel = 0,
        LayoutOrder = order,
    }, parent)
    corner(row, 8)

    new("TextLabel", {
        Position = UDim2.new(0, 12, 0, 8),
        Size = UDim2.new(0.55, 0, 0, 16),
        BackgroundTransparency = 1,
        Text = label,
        Font = Enum.Font.GothamMedium,
        TextSize = 13,
        TextColor3 = Theme.Text,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)

    local valueBox = new("TextBox", {
        Position = UDim2.new(0.55, 0, 0, 6),
        Size = UDim2.new(0.45, -12, 0, 20),
        BackgroundColor3 = Color3.fromRGB(24, 25, 29),
        BorderSizePixel = 0,
        Text = tostring(default),
        Font = Enum.Font.GothamBold,
        TextSize = 12,
        TextColor3 = Theme.Accent,
        TextXAlignment = Enum.TextXAlignment.Center,
        ClearTextOnFocus = false,
    }, row)
    corner(valueBox, 6)

    local track = new("Frame", {
        Position = UDim2.new(0, 12, 0, 36),
        Size = UDim2.new(1, -24, 0, 4),
        BackgroundColor3 = Color3.fromRGB(55, 57, 64),
        BorderSizePixel = 0,
    }, row)
    corner(track, 2)

    local ratio = math.clamp(default / maxValue, 0, 1)
    local fill = new("Frame", {
        Size = UDim2.new(ratio, 0, 1, 0),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
    }, track)
    corner(fill, 2)

    local knob = new("Frame", {
        Size = UDim2.fromOffset(12, 12),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(ratio, 0, 0.5, 0),
        BackgroundColor3 = Color3.fromRGB(240, 240, 245),
        BorderSizePixel = 0,
        ZIndex = 2,
    }, track)
    corner(knob, 6)

    local hit = new("TextButton", {
        Size = UDim2.new(1, 16, 3, 16),
        Position = UDim2.new(0, -8, 0.5, 0),
        AnchorPoint = Vector2.new(0, 0.5),
        BackgroundTransparency = 1,
        Text = "",
        ZIndex = 3,
    }, track)

    local dragging = false
    local focused  = false
    local currentValue = default

    local function setValue(value, updateBox)
        value = math.clamp(math.floor(value + 0.5), 0, maxValue)
        currentValue = value
        local rel = value / maxValue
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, 0, 0.5, 0)
        if updateBox and not focused then
            valueBox.Text = tostring(value)
        end
        if onChange then onChange(value) end
    end

    setValue(default, false)

    local function applyFromX(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
        local value = math.floor(rel * maxValue + 0.5)
        setValue(value, true)
    end

    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            applyFromX(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            applyFromX(input.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    valueBox.Focused:Connect(function()
        focused = true
    end)
    valueBox.FocusLost:Connect(function()
        focused = false
        local num = tonumber(valueBox.Text)
        if not num then
            valueBox.Text = tostring(currentValue)
            return
        end
        num = math.floor(num + 0.5)
        if num > maxValue then num = maxValue end
        if num < 0 then num = 0 end
        setValue(num, true)
    end)
end

local GeneralState = {
    SpeedValue   = 16,
    SpeedKeep    = false,
    JumpValue    = 50,
    JumpKeep     = false,
    InfiniteJump = false,
}

local function applySpeed()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = GeneralState.SpeedValue end
end

local function applyJump()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.UseJumpPower = true
        hum.JumpPower = GeneralState.JumpValue
    end
end

task.spawn(function()
    while task.wait(0.1) do
        if GeneralState.SpeedKeep then applySpeed() end
        if GeneralState.JumpKeep  then applyJump()  end
    end
end)

    if not GeneralState.InfiniteJump then return end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position

            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    conn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == Enum.KeyCode.RightShift then
            if gui.Enabled and Main.Visible then
                closeUI()
            else
                gui.Enabled = true
                Island.Visible = false
                StatsPill.Visible = false
                Main.Visible = true
            end
        end
    end)
end
