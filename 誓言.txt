-- ==================== 基础工具 ====================
local lockPlayerConn = nil
local function uprightCFrame(hrp, cf)
    local pos = cf.Position
    local yRot = hrp and hrp.Orientation.Y or 0
    return CFrame.new(pos) * CFrame.Angles(0, math.rad(yRot), 0)
end
local function unlockPlayer()
    if lockPlayerConn then lockPlayerConn:Disconnect(); lockPlayerConn = nil end
end
local function lockPlayerAt(cf)
    unlockPlayer()
    local function hold()
        local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.CFrame = uprightCFrame(hrp, cf)
            hrp.Velocity = Vector3.new(0,0,0)
            hrp.RotVelocity = Vector3.new(0,0,0)
        end
    end
    hold()
    lockPlayerConn = RS.Heartbeat:Connect(hold)
end
local function safeTeleport(cf)
    local hrp = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.CFrame = uprightCFrame(hrp, cf)
        task.wait(0.05)
        if hrp.Position.Y < -100 then hrp.CFrame = uprightCFrame(hrp, cf) + Vector3.new(0,10,0) end
    end
end

-- ==================== 强拖拽（含到位校验） ====================
local DRAG_TOTAL = 60
local DRAG_SET_START = 28
local DRAG_REFRESH_INTERVAL = 0.01

local function dragToPosition(model, targetCF)
    if not model or not model.Parent then return false end
    local primary = model.PrimaryPart
        or model:FindFirstChild("WoodSection")
        or model:FindFirstChild("Main")
        or model:FindFirstChildWhichIsA("BasePart")
    if not primary then return false end
    if not model.PrimaryPart then model.PrimaryPart = primary end

    lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))

    local remote = rep.Interaction.ClientIsDragging
    local dragID = math.random(100000, 999999)

    pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(primary) end)
    pcall(function() remote:FireServer("Begin", model, dragID) end)

    for i = 1, 30 do
        if not model.Parent or not model.PrimaryPart then
            pcall(function() remote:FireServer("End", model, dragID) end)
            unlockPlayer(); return false
        end
        pcall(function()
            remote:FireServer("Refresh", model, dragID)
            if i >= DRAG_SET_START then
                rep.Interaction.ClientRequestOwnership:FireServer(primary)
                model:SetPrimaryPartCFrame(targetCF)
            end
        end)
        task.wait(DRAG_REFRESH_INTERVAL)
    end
    for _ = 31, DRAG_TOTAL do
        if not model.Parent then break end
        pcall(function() remote:FireServer("Refresh", model, dragID) end)
        task.wait(DRAG_REFRESH_INTERVAL)
    end
    pcall(function() remote:FireServer("End", model, dragID) end)
    task.wait(0.1)

    -- 到位校验 + 补偿
    local targetPos = targetCF.Position
    for _ = 1, 3 do
        local pp = model.PrimaryPart or model:FindFirstChild("WoodSection") or model:FindFirstChild("Main")
        if not pp then break end
        if (pp.Position - targetPos).Magnitude <= 5 then break end
        lockPlayerAt(pp.CFrame + Vector3.new(0, 3, 0))
        task.wait(0.15)
        local retryID = math.random(100000, 999999)
        pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(pp) end)
        pcall(function() remote:FireServer("Begin", model, retryID) end)
        for _ = 1, 15 do
            if not model.Parent then break end
            pcall(function() rep.Interaction.ClientRequestOwnership:FireServer(pp) end)
            pcall(function() remote:FireServer("Refresh", model, retryID) end)
            pcall(function() model:SetPrimaryPartCFrame(targetCF) end)
            task.wait(0.03)
        end
        pcall(function() remote:FireServer("End", model, retryID) end)
        task.wait(0.15)
    end

    unlockPlayer()
    return true
end

local function readTreeClasses()
    local list = {}
    local amounts = ws:FindFirstChild("Stores")
        and ws.Stores:FindFirstChild("PlanterStore")
        and ws.Stores.PlanterStore:FindFirstChild("Planter Spawner")
        and ws.Stores.PlanterStore["Planter Spawner"]:FindFirstChild("Amounts")
    if amounts then
        for _, c in ipairs(amounts:GetChildren()) do
            if c:IsA("NumberValue") or c:IsA("IntValue") or c:IsA("StringValue") then
                table.insert(list, tostring(c.Name))
            end
        end
    end
    if #list == 0 then
        local W = rep:FindFirstChild("Woods")
        if W then
            for _, c in ipairs(W:GetChildren()) do table.insert(list, tostring(c.Name)) end
        end
    end
    return list
end

local TREE_PRIORITY = {"火焰木","石头木","蓝色火焰木","彩色火焰木"}
local function sortTreeNames(list)
    local prio = {}
    for i, n in ipairs(TREE_PRIORITY) do prio[n] = i end
    table.sort(list, function(a, b)
        local pa, pb = prio[a], prio[b]
        if pa and pb then return pa < pb end
        if pa then return true end
        if pb then return false end
        return a < b
    end)
end
local treeNames = {}
for name in pairs(treeMapping) do table.insert(treeNames, name) end
sortTreeNames(treeNames)

-- ==================== 斧头配置（decompile） ====================
local axeConfigCache = {}
local function loadWeaponConfig(weaponName)
    if axeConfigCache[weaponName] ~= nil then return axeConfigCache[weaponName] end

    -- 先尝试 AxeClassDamageOverride（OlReliable 特有）
    if weaponName == "OlReliable" or weaponName == "OIReliable" then
        local override = nil
        local wsTool = workspace:FindFirstChild(lp.Name)
        if wsTool then
            local t = wsTool:FindFirstChild("Tool")
            if t then
                local tn = t:FindFirstChild("ToolName")
                if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                    override = t:FindFirstChild("AxeClassDamageOverride")
                end
            end
        end
        if not override then
            local bp = lp:FindFirstChild("Backpack")
            if bp then
                for _, tool in pairs(bp:GetChildren()) do
                    if tool:IsA("Tool") then
                        local tn = tool:FindFirstChild("ToolName")
                        if tn and (tostring(tn.Value)=="OlReliable" or tostring(tn.Value)=="OIReliable") then
                            override = tool:FindFirstChild("AxeClassDamageOverride")
                            if override then break end
                        end
                    end
                end
            end
        end
        if override then
            local cfg = {Damage=nil, SwingCooldown=nil, Range=nil, SpecialTrees={}}
            for k, v in pairs(override:GetAttributes()) do
                if k == "Damage" then cfg.Damage = v
                elseif k == "SwingCooldown" then cfg.SwingCooldown = v
                elseif k == "Range" then cfg.Range = v
                else cfg.SpecialTrees[k] = {Damage = v} end
            end
            for _, c in pairs(override:GetChildren()) do
                if c:IsA("NumberValue") or c:IsA("IntValue") then
                    if c.Name == "Damage" then cfg.Damage = c.Value
                    elseif c.Name == "SwingCooldown" then cfg.SwingCooldown = c.Value
                    elseif c.Name == "Range" then cfg.Range = c.Value
                    else cfg.SpecialTrees[c.Name] = {Damage = c.Value} end
                end
            end
            if cfg.Damage or cfg.SwingCooldown or next(cfg.SpecialTrees) then
                axeConfigCache[weaponName] = cfg
                return cfg
            end
        end
    end

    -- 反编译 AxeClass
    local function decompileSafe(s)
        if not s then return nil end
        if type(decompile) == "function" then
            local ok, src = pcall(decompile, s)
            if ok and type(src) == "string" and #src > 0 then return src end
        end
        if type(getscriptbytecode) == "function" then
            local ok, bc = pcall(getscriptbytecode, s)
            if ok and type(bc) == "string" then return bc end
        end
        return nil
    end
    local function parseNum(src, key)
        local v = src:match("[%w_%.]+%.?" .. key .. "%s*=%s*([%-]?[%d%.]+)")
        if not v then v = src:match(key .. "%s*=%s*([%-]?[%d%.]+)") end
        return tonumber(v)
    end
    local function parseSpecialTrees(src)
        local out = {}
        for treeName, body in src:gmatch("SpecialTrees%.([%w_]+)%s*=%s*{(.-)}") do
            local cfg = {}
            cfg.Damage = tonumber(body:match("Damage%s*=%s*([%-]?[%d%.]+)"))
            cfg.SwingCooldown = tonumber(body:match("SwingCooldown%s*=%s*([%-]?[%d%.]+)"))
            if cfg.Damage then out[treeName] = cfg end
        end
        return out
    end
    local function findAxeModule(name)
        local want = name .. "AxeClass"
        local la = rep:FindFirstChild("LoadedAssets")
        if la then local m = la:FindFirstChild(want); if m then return m end end
        if type(getloadedmodules) == "function" then
            for _, m in ipairs(getloadedmodules()) do
                if m:IsA("ModuleScript") and (m.Name == want or m:GetFullName():sub(-#want) == want) then
                    return m
                end
            end
        end
        return nil
    end

    local module = findAxeModule(weaponName)
    if not module then
        axeConfigCache[weaponName] = nil
        return nil
    end
    local src = decompileSafe(module)
    if not src then
        axeConfigCache[weaponName] = nil
        return nil
    end
    local cfg = {
        Damage = parseNum(src, "Damage"),
        SwingCooldown = parseNum(src, "SwingCooldown"),
        Range = parseNum(src, "Range"),
        SpecialTrees = parseSpecialTrees(src),
    }
    if not cfg.Damage or not cfg.SwingCooldown then
        axeConfigCache[weaponName] = nil
        return nil
    end
    cfg.Range = cfg.Range or 10
    axeConfigCache[weaponName] = cfg
    return cfg
end

local function selectBestWeapon(treeClass)
    local tf = lp:FindFirstChild("ToolFolder")
    if not tf then return nil, nil, "未找到 ToolFolder" end
    local bestName, bestDmg, bestCfg = nil, -1, nil
    for _, child in pairs(tf:GetChildren()) do
        local cfg = loadWeaponConfig(child.Name)
        if cfg then
            local dmg = (cfg.SpecialTrees[treeClass] and cfg.SpecialTrees[treeClass].Damage) or cfg.Damage
            if dmg > bestDmg then bestName, bestDmg, bestCfg = child.Name, dmg, cfg end
        end
    end
    return bestName, bestCfg, bestDmg
end

local function getBestAxeData(treeClass)
    local tf = lp:FindFirstChild("ToolFolder")
    if not tf then return nil, nil, "未找到 ToolFolder" end
    local bestName, bestDmg, bestCfg = nil, -1, nil
    for _, child in pairs(tf:GetChildren()) do
        local cfg = loadWeaponConfig(child.Name)
        if cfg then
            local dmg = cfg.Damage or 0
            if treeClass and cfg.SpecialTrees and cfg.SpecialTrees[treeClass] then
                dmg = cfg.SpecialTrees[treeClass].Damage or dmg
            end
            if dmg > bestDmg then bestDmg = dmg; bestName = child.Name; bestCfg = cfg end
        end
    end
    if not bestName then return nil, nil, "未找到可用的斧头配置" end
    local obj = tf:FindFirstChild(bestName)
    local tool = obj and (obj:IsA("Tool") and obj or obj.Value)
    if not tool then return nil, nil, "无法获取工具实例" end
    return tool, bestCfg, nil
end

-- ==================== 找树 ====================
local function findUncutTree(treeClass)
    local playerPos = lp.Character and lp.Character.HumanoidRootPart.Position
    if not playerPos then return nil end
    local nearestDist, nearestWood, nearestTree = math.huge, nil, nil
    for _, region in pairs(ws:GetChildren()) do
        if region.Name == "TreeRegion" then
            for _, tree in pairs(region:GetChildren()) do
                if tree:IsA("Model") and tree:FindFirstChild("TreeClass") and tree.TreeClass.Value == treeClass then
                    local sections = {}
                    for _, c in pairs(tree:GetChildren()) do
                        if c.Name == "WoodSection" and c:IsA("BasePart") then table.insert(sections, c) end
                    end
                    if #sections >= 2 then
                        local d = (sections[1].Position - playerPos).Magnitude
                        if d < nearestDist then nearestDist, nearestWood, nearestTree = d, sections[1], tree end
                    end
                end
            end
        end
    end
    return nearestWood, nearestTree
end
local function getTreeClassFromModel(m)
    local tc = m:FindFirstChild("TreeClass")
    if tc then return tc.Value end
    local n = m.Name
    for _, s in ipairs({"SpookyNeon","Spooky","GoldSwampy","Cherry","CaveCrawler","MuckySewer"}) do
        if n:find(s) then return s end
    end
    return "Generic"
end
local function getTreeTopCFrame(tree)
    local topY, topCF = -math.huge, nil
    for _, c in pairs(tree:GetChildren()) do
        if c:IsA("BasePart") then
            local top = c.Position.Y + c.Size.Y / 2
            if top > topY then
                topY = top
                topCF = CFrame.new(c.Position.X, top + 2, c.Position.Z)
            end
        end
    end
    return topCF or (tree:GetPivot() + Vector3.new(0, 2, 0))
end

-- ==================== 带来树 ====================
local function bringTreeRemote(treeClass, targetCF)
    local weaponName, weaponConfig, damage = selectBestWeapon(treeClass)
    if not weaponName then notify("誓言", "自动选武器失败", 3); return false end
    local req = {LoveCave=10000000, Shine=80000000, Magma=5000000, Ice=8000000,
                 Radioactive=10000000, Scale=40000000, Void=5000000, Flame=10000000,
                 BlueFlame=3500000, RainbowFlame=3500000, Celestial=10000000}
    if req[treeClass] and damage < req[treeClass] then
        notify("誓言", string.format("伤害不足！需要至少 %.0f", req[treeClass]), 4)
        return false
    end
    local cooldown = (weaponConfig.SpecialTrees[treeClass] and weaponConfig.SpecialTrees[treeClass].SwingCooldown) or weaponConfig.SwingCooldown

    local woodSec, tree = findUncutTree(treeClass)
    if not woodSec then notify("誓言", "附近没有可砍的树", 3); return false end

    local tool = lp.Character and lp.Character:FindFirstChildOfClass("Tool")
    if not tool then
        local tf = lp:FindFirstChild("ToolFolder")
        if tf then
            local wo = tf:FindFirstChild(weaponName)
            tool = wo and (wo:IsA("Tool") and wo or wo.Value) or nil
        end
    end
    if not tool then notify("誓言", "未找到工具: " .. weaponName, 3); return false end

    local cutEvent = tree:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "无法获取 CutEvent", 3); return false end
    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")

    if not targetCF then
        targetCF = lp.Character and lp.Character.HumanoidRootPart.CFrame
        if not targetCF then notify("誓言", "无法获取目标位置", 3); return false end
    end

    lockPlayerAt(getTreeTopCFrame(tree))

    local fallenLog, existing = nil, {}
    for _, log in pairs(ws.LogModels:GetChildren()) do existing[log] = true end
    local treeCenter = woodSec.Position
    local stopCutting, timeoutReached = false, false

    local sendTask = task.spawn(function()
        while not stopCutting do
            pcall(remote.FireServer, remote, cutEvent, {
                height=0.3, faceVector=Vector3.new(-1,0,0), cooldown=cooldown,
                sectionId=1, hitPoints=damage, tool=tool, cuttingClass="Axe"
            })
            task.wait(0.001)
        end
    end)
    local timeoutTask = task.spawn(function()
        task.wait(15)
        if not stopCutting then timeoutReached = true; stopCutting = true end
    end)

    while not fallenLog and not stopCutting do
        for _, log in pairs(ws.LogModels:GetChildren()) do
            if not existing[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp
            and log:FindFirstChild("TreeClass") and log.TreeClass.Value == treeClass then
                if (log:GetPivot().Position - treeCenter).Magnitude < 100 then
                    fallenLog = log; break
                end
            end
        end
        task.wait(0.01)
    end
    stopCutting = true
    task.cancel(sendTask); task.cancel(timeoutTask)
    unlockPlayer()

    if timeoutReached or not fallenLog then notify("誓言", "砍树超时", 3); return false end

    local primary = fallenLog.PrimaryPart or fallenLog:FindFirstChild("WoodSection")
        or fallenLog:FindFirstChild("Main") or fallenLog:FindFirstChildWhichIsA("BasePart")
    if not primary then notify("誓言", "无法定位木头", 3); return false end
    if not fallenLog.PrimaryPart then fallenLog.PrimaryPart = primary end

    lockPlayerAt(primary.CFrame + Vector3.new(0, 3, 0))
    task.wait(0.15)
    for _ = 1, 3 do
        pcall(rep.Interaction.ClientRequestOwnership.FireServer, rep.Interaction, primary)
        task.wait(0.01)
    end

    dragToPosition(fallenLog, targetCF)
    safeTeleport(targetCF)
    unlockPlayer()
    notify("誓言", "树已带来", 2)
    return true
end

local function selectSawmill()
    if isSelectingSawmill then notify("誓言", "正在等待选择中...", 2); return end
    isSelectingSawmill = true
    notify("誓言", "请点击锯木机任意部件", 3)
    if sawmillSelectConn then sawmillSelectConn:Disconnect() end
    sawmillSelectConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then notify("誓言", "未选中任何物体", 2); return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then notify("誓言", "请点击一个模型", 2); return end

        local function isSawmill(root)
            local attr = root:GetAttribute("ItemName")
            if attr and type(attr) == "string" and attr:lower():find("saw") then return true end
            for _, c in pairs(root:GetDescendants()) do
                if c.Name == "ItemName" then
                    local v = c.Value
                    if v and type(v) == "string" and v:lower():find("saw") then return true end
                end
            end
            if root.Name and root.Name:lower():find("saw") then return true end
            local tv = root:FindFirstChild("Type")
            if tv and tv.Value == "Sawmill" then return true end
            return false
        end
        if not isSawmill(model) then notify("誓言", "这不是锯木机", 3); return end

        local function findPart(root)
            for _, c in pairs(root:GetDescendants()) do
                if c.Name == "Conveyor" and c:IsA("BasePart") then return c, "Conveyor" end
            end
            for _, c in pairs(root:GetDescendants()) do
                if c.Name == "Main" and c:IsA("BasePart") then return c, "Main" end
            end
            return nil, nil
        end
        local part, pt = findPart(model)
        if not part then notify("誓言", "找不到 Conveyor 或 Main 部件", 3); return end

        cachedSawmill = model
        cachedSawmillPos = part.CFrame
        if sawmillSelectConn then sawmillSelectConn:Disconnect(); sawmillSelectConn = nil end
        isSelectingSawmill = false
        notify("誓言", "✅ 已记忆锯木机位置（" .. pt .. "）", 3)
    end)
end

-- ==================== 一键加工 ====================
local isDecomposeActive = false
local decomposeConn = nil

local function decomposeAndProcess(treeModel, sawmillPos)
    if not sawmillPos then notify("誓言", "锯木机位置无效", 3); return end
    local sections, rootSection = {}, nil
    for _, c in pairs(treeModel:GetChildren()) do
        if c.Name == "WoodSection" and c:IsA("BasePart") and c:FindFirstChild("ID") then
            local id = c.ID.Value
            if id >= 2 then table.insert(sections, {id=id, part=c})
            elseif id == 1 then rootSection = c end
        end
    end
    if #sections == 0 then notify("誓言", "该树没有可分解的部分", 3); return end
    table.sort(sections, function(a, b) return a.id > b.id end)

    local cutEvent = treeModel:FindFirstChild("CutEvent")
    if not cutEvent then notify("誓言", "该模型没有 CutEvent", 3); return end

    local treeClass = getTreeClassFromModel(treeModel)
    local tool, cfg, err = getBestAxeData(treeClass)
    if not tool then notify("誓言", err or "无法获取斧头数据", 3); return end

    local damage = cfg and cfg.Damage or 10000
    local cooldown = cfg and cfg.SwingCooldown or 0.3
    if cfg and cfg.SpecialTrees and cfg.SpecialTrees[treeClass] then
        damage = cfg.SpecialTrees[treeClass].Damage or damage
        cooldown = cfg.SpecialTrees[treeClass].SwingCooldown or cooldown
    end

    local req = {LoveCave=10000000, Shine=80000000, Magma=5000000, Ice=8000000,
                 Radioactive=10000000, Scale=40000000, Void=5000000, Flame=10000000,
                 BlueFlame=3500000, RainbowFlame=3500000, Celestial=10000000}
    if req[treeClass] and damage < req[treeClass] then
        notify("誓言", string.format("伤害不足！需要至少 %.0f，当前 %.0f", req[treeClass], damage), 4)
        return
    end

    local remote = rep:WaitForChild("Interaction"):WaitForChild("RemoteProxy")
    local initialLogs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do initialLogs[log] = true end

    local tasks = {}
    for _, sec in ipairs(sections) do
        local part = sec.part
        local sectionId = sec.id
        local stopFlag = false
        local taskId = task.spawn(function()
            while not stopFlag do
                pcall(remote.FireServer, remote, cutEvent, {
                    height=0.11, faceVector=Vector3.new(-1,0,0), cooldown=cooldown,
                    sectionId=sectionId, hitPoints=damage, tool=tool, cuttingClass="Axe"
                })
                task.wait(0.01)
                if not part or not part.Parent then break end
            end
        end)
        tasks[taskId] = true
    end

    local function hasHighID()
        for _, c in pairs(treeModel:GetChildren()) do
            if c.Name == "WoodSection" and c:FindFirstChild("ID") and c.ID.Value >= 2 then
                return true
            end
        end
        return false
    end

    local startTime = tick()
    while hasHighID() and tick() - startTime < 120 do task.wait(0.01) end
    for tid in pairs(tasks) do task.cancel(tid) end
    if hasHighID() then notify("誓言", "加工超时", 3); return end

    local logs = {}
    for _, log in pairs(ws.LogModels:GetChildren()) do
        if not initialLogs[log] and log:FindFirstChild("Owner") and log.Owner.Value == lp then
            table.insert(logs, log)
        end
    end
    if #logs == 0 then notify("誓言", "未找到木头", 3); return end

    local targetPos = sawmillPos + Vector3.new(0.7, 0, 0)
    for _, log in ipairs(logs) do
        if log and log.Parent then dragToPosition(log, targetPos) end
    end
    if rootSection and rootSection.Parent then
        if not treeModel.PrimaryPart then treeModel.PrimaryPart = rootSection end
        safeTeleport(rootSection.CFrame + Vector3.new(0, 2, 0))
        dragToPosition(treeModel, targetPos)
    end
    notify("誓言", "分解完成", 3)
end

local function startDecomposeSelection()
    if isDecomposeActive then notify("誓言", "分解选择已激活", 2); return end
    if not cachedSawmill or not cachedSawmill.Parent then
        notify("誓言", "请先使用「选择锯木机」按钮", 3); return
    end
    if not cachedSawmillPos then notify("誓言", "锯木机位置无效", 3); return end
    isDecomposeActive = true
    notify("誓言", "请点击一棵树", 3)
    if decomposeConn then decomposeConn:Disconnect() end
    decomposeConn = mouse.Button1Down:Connect(function()
        local target = mouse.Target
        if not target then notify("誓言", "未选中任何物体", 2); return end
        local model = target:FindFirstAncestorOfClass("Model")
        if not model then notify("誓言", "请点击树干", 2); return end
        local hasWood = false
        for _, c in pairs(model:GetChildren()) do
            if c.Name == "WoodSection" then hasWood = true end
        end
        if not (hasWood and model:FindFirstChild("CutEvent")) then
            notify("誓言", "这不是一棵树", 2); return
        end
        if decomposeConn then decomposeConn:Disconnect(); decomposeConn = nil end
        isDecomposeActive = false
        notify("誓言", "已选中树，开始分解...", 3)
        task.spawn(function()
            decomposeAndProcess(model, cachedSawmillPos)
        end)
    end)
end
